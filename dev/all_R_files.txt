test_that(".normalize_or_generate_batch_id uses explicit id when provided", {
  expect_equal(
    .normalize_or_generate_batch_id("stream_a", "method_b", "explicit_123"),
    "explicit_123"
  )

test_that(".normalize_or_generate_batch_id generates predictable format", {
  gen_id <- .normalize_or_generate_batch_id("stream_a", "method_b")

  expect_type(gen_id, "character")
  expect_true(grepl("_stream_a_method_b$", gen_id))
})

test_that(".normalize_or_generate_batch_id rejects an empty explicit batch_id", {
  expect_error(
    .normalize_or_generate_batch_id("s", "m", ""),
    class = "xpipe_batch_id"
  )
})

test_that(".normalize_or_generate_batch_id rejects a non-string explicit batch_id", {
  # character(0) fails the rlang::check_string guard before reaching nzchar
  expect_error(
    .normalize_or_generate_batch_id("s", "m", character(0))
  )
})

# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
# test-batch_materialization.R
#
# Tests for the materialization pipeline internals, covering the try_fetch
# error-wrapping in .materialize_batch, instrument resolution, and the
# extraction/persistence phases.

# ---- .resolve_extraction_instrument -----------------------------------------

test_that(".resolve_extraction_instrument aborts on unregistered method type", {
  ctx <- list(method_config = list(type = "unknown_format"))

  expect_error(
    .resolve_extraction_instrument(ctx, NULL),
    class = "xpipe_batch_instrument"
  )
})

test_that(".resolve_extraction_instrument aborts when function is not loaded", {
  # Register a type that maps to a function that doesn't exist in the env
  ctx <- list(method_config = list(type = "access_mdb"))

  # Temporarily ensure the function is absent
  if (exists("extract_access_db", mode = "function")) {
    skip("extract_access_db is loaded; cannot test missing-function guard here")
  }

  expect_error(
    .resolve_extraction_instrument(ctx, NULL),
    class = "xpipe_batch_instrument"
  )
})

# ---- .run_extraction --------------------------------------------------------

test_that(".run_extraction wraps a failing single-file instrument as xpipe_batch_extraction", {
  failing_instrument <- function(ctx, path) stop("ODBC: connection refused")

  ctx <- list(
    components = NULL,
    filepath = tempfile(),
    method_config = list(type = "access_mdb")
  )

  # Reproduce the try_fetch logic that .materialize_batch applies
  expect_error(
    rlang::try_fetch(
      .run_extraction(ctx, failing_instrument),
      error = function(e) cli::cli_abort(
        "Data extraction failed",
        parent = e,
        class = c("xpipe_batch_extraction", "xpipe_batch_error", "xpipe_error")
      )
    ),
    class = "xpipe_batch_extraction"
  )
})

test_that(".run_extraction aborts when extractor returns an empty list", {
  empty_instrument <- function(ctx, path) list()

  ctx <- list(
    components = NULL,
    filepath = tempfile(),
    method_config = list(type = "access_mdb")
  )

  expect_error(
    .run_extraction(ctx, empty_instrument),
    class = "xpipe_batch_extraction"
  )
})

test_that(".run_extraction coerces a bare data.frame return to a named list", {
  df_instrument <- function(ctx, path) data.frame(x = 1:3)

  ctx <- list(
    components = NULL,
    filepath = tempfile(),
    method_config = list(type = "access_mdb")
  )

  result <- .run_extraction(ctx, df_instrument)
  expect_type(result, "list")
  expect_named(result, "data")
  expect_s3_class(result$data, "data.frame")
})

test_that(".run_extraction aborts if an extracted element is not a data.frame", {
  bad_instrument <- function(ctx) list(good_table = data.frame(x = 1), bad_table = "not a df")

  ctx <- list(
    components = NULL,
    filepath = tempfile(),
    method_config = list(type = "access_mdb")
  )

  expect_error(
    .run_extraction(ctx, bad_instrument),
    class = "xpipe_batch_extraction"
  )
})

test_that(".run_extraction aborts if extracted list is not fully named", {
  unnamed_instrument <- function(ctx) list(data.frame(x = 1))

  ctx <- list(
    components = NULL,
    filepath = tempfile(),
    method_config = list(type = "access_mdb")
  )

  expect_error(
    .run_extraction(ctx, unnamed_instrument),
    class = "xpipe_batch_extraction"
  )
})

# ---- .persist_to_batch ------------------------------------------------------

test_that(".persist_to_batch wraps filesystem errors as xpipe_batch_materialization", {
  # Point batch_dir at a path that cannot be created (a file used as a dir)
  tmp_file <- tempfile()
  writeLines("block", tmp_file)  # exists as a file, not a directory

  ctx <- list(
    batch_id = "test_batch",
    batch_dir = fs::path(tmp_file, "subdir"),  # impossible path
    filename = "test.mdb",
    stream_id = "sdo",
    method_id = "sdo_quani_mdb",
    method_config = list(
      schema_domain = "sdo",
      type = "access_mdb",
      input_type = "file"
    ),
    stream_config = list(anchor = "start")
  )

  mock_config <- structure(
    list(runtime = list(tool = list(name = "xpipe", version = "dev"))),
    class = "xpipe_config"
  )

  good_data <- list(TblScheda = data.frame(x = 1:3))

  expect_error(
    rlang::try_fetch(
      .persist_to_batch(ctx, good_data, mock_config),
      error = function(e) cli::cli_abort(
        "Failed to persist batch data",
        parent = e,
        class = c(
          "xpipe_batch_materialization",
          "xpipe_batch_error",
          "xpipe_error"
        )
      )
    ),
    class = "xpipe_batch_materialization"
  )
})

# This tests that the environment variables and global state work
# correctly without actually messing with your real environment.

test_that("Runtime context aborts if pepper is missing", {
  withr::with_envvar(c("XNE_PEPPER_KEY" = NA_character_), {
    expect_error(
      .load_runtime_context(),
      class = "xpipe_runtime_pepper"
    )
  })
})

test_that("Runtime context loads correctly when pepper is present", {
  withr::with_envvar(c("XNE_PEPPER_KEY" = "fake_secret_key"), {
    ctx <- .load_runtime_context()
    expect_equal(ctx$pepper, "fake_secret_key")
    expect_true(inherits(ctx$loaded_at, "POSIXct"))
  })
})

test_that("Package environment stores and retrieves config", {
  mock_config <- structure(list(mock = TRUE), class = "xpipe_config")

  # Inject into the package environment
  .xpipe_env$config <- mock_config
  retrieved <- get_xpipe_config()
  expect_true(retrieved$mock)

  # Clear it and test the specific error class
  .xpipe_env$config <- NULL
  expect_error(
    get_xpipe_config(),
    class = "xpipe_config_missing"
  )
})

tests/testthat/test-crypto.R
'''
test_that("hash_phi hashes deterministically and ignores case/whitespace", {
  pepper <- "test_pepper_123"
  val1 <- hash_phi("John Doe", pepper)
  val2 <- hash_phi("JOHN DOE", pepper)
  val3 <- hash_phi("  John Doe  ", pepper)

  expect_type(val1, "character")
  expect_equal(val1, val2)
  expect_equal(val1, val3)
})

test_that("hash_phi handles NAs and empty strings", {
  pepper <- "test_pepper_123"
  expect_equal(hash_phi(NA_character_, pepper), NA_character_)
  expect_equal(hash_phi("", pepper), NA_character_)

  # Vectorized NA handling
  res <- hash_phi(c("A", NA, ""), pepper)
  expect_equal(is.na(res), c(FALSE, TRUE, TRUE))
})

test_that("hash_phi aborts on missing or empty pepper", {
  expect_error(
    hash_phi("John Doe", ""),
    class = "xpipe_runtime_pepper"
  )
  # Missing argument also fails the same guard
  expect_error(
    hash_phi("John Doe", NA_character_),
    class = "xpipe_runtime_pepper"
  )
})

# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
# test-guardrails.R
# Tests for pre-materialization guardrails (.assert_batch_materializable).
# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

test_that(".assert_batch_materializable catches collisions and missing inputs", {
  mock_config <- structure(list(), class = "xpipe_config")

  withr::with_tempdir({
    ctx <- list(
      batch_id  = "test_batch_1",
      batch_dir = fs::path_abs("data/batch/test_batch_1"),
      filepath  = fs::path_abs("inbox/test_file.csv")
    )

    # Test A: target directory already exists (collision)
    fs::dir_create(ctx$batch_dir)
    expect_error(
      .assert_batch_materializable(ctx, mock_config),
      class = "xpipe_batch_collision"
    )
    expect_error(
      .assert_batch_materializable(ctx, mock_config),
      "already exists"
    )

    fs::dir_delete(ctx$batch_dir)

    # Test B: input file does not exist at materialization time
    expect_error(
      .assert_batch_materializable(ctx, mock_config),
      class = "xpipe_batch_input"
    )
    expect_error(
      .assert_batch_materializable(ctx, mock_config),
      "does not exist at materialization time"
    )

    # Test C: input file exists but is empty (0 bytes)
    fs::dir_create("inbox")
    fs::file_create(ctx$filepath)
    expect_error(
      .assert_batch_materializable(ctx, mock_config),
      class = "xpipe_batch_input"
    )
    expect_error(
      .assert_batch_materializable(ctx, mock_config),
      "empty or unreadable"
    )

    # Test D: valid state — file exists and has content
    writeLines("test data", ctx$filepath)
    expect_invisible(.assert_batch_materializable(ctx, mock_config))
  })
})

# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
# test-inspect.R
# Tests for Batch layer inspection and formatting utilities.
# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

test_that("inspect_batch_layer correctly identifies complete and incomplete batches", {
  mock_config <- structure(
    list(local = list(paths = list(batch = "mock_batch_layer"))),
    class = "xpipe_config"
  )

  withr::with_tempdir({
    fs::dir_create("mock_batch_layer")

    # 1. Create a COMPLETE batch with valid metadata
    fs::dir_create("mock_batch_layer/batch_healthy")
    jsonlite::write_json(
      list(
        batch_id  = "batch_healthy",
        stream    = list(id = "sdo", schema_domain = "sdo"),
        method    = list(id = "xml_campania_2026", type = "xml_sdo_trac_b"),
        input     = list(path = "test.xml"),
        execution = list(created_at = "2026-04-30T10:00:00Z", tool = "xpipe")
      ),
      "mock_batch_layer/batch_healthy/_batch_metadata.json",
      auto_unbox = TRUE
    )

    # 2. Create an INCOMPLETE batch (no metadata file)
    fs::dir_create("mock_batch_layer/batch_corrupt")

    inspection <- inspect_batch_layer(mock_config)

    expect_s3_class(inspection, "xpipe_batch_layer")
    expect_equal(inspection$count, 2)
    expect_equal(inspection$batches$batch_corrupt$.status, "incomplete")
    expect_equal(inspection$batches$batch_healthy$batch_id, "batch_healthy")
    expect_null(inspection$batches$batch_healthy$.status)
  })
})

test_that("print.xpipe_batch_layer does not crash when all batches are incomplete", {
  mock_config <- structure(
    list(local = list(paths = list(batch = "mock_batch_layer"))),
    class = "xpipe_config"
  )

  withr::with_tempdir({
    fs::dir_create("mock_batch_layer")
    fs::dir_create("mock_batch_layer/broken_1")
    fs::dir_create("mock_batch_layer/broken_2")

    inspection <- inspect_batch_layer(mock_config)

    # Printing should not throw
    expect_error(suppressMessages(print(inspection)), NA)

    res <- withVisible(suppressMessages(print(inspection)))
    expect_false(res$visible)
    expect_identical(res$value, inspection)
  })
})

test_that("inspect_batch_layer aborts if batch layer path does not exist", {
  mock_config <- structure(
    list(local = list(paths = list(batch = "path_that_does_not_exist"))),
    class = "xpipe_config"
  )

  expect_error(
    inspect_batch_layer(mock_config),
    class = "xpipe_batch_paths"
  )
})

test_that("inspect_batch_layer aborts if config is not an xpipe_config", {
  expect_error(
    inspect_batch_layer(list()),
    class = "xpipe_batch_config"
  )
})
