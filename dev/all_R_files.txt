# Error Class Taxonomy --------------------------------------------------------
#
# All errors inherit from `xpipe_error` (package base) and `xpipe_batch_error`
# (file-level domain). New leaves from core_config.R taxonomy also apply here
# (xpipe_batch_paths, xpipe_batch_metadata, xpipe_batch_hash).
#
#   xpipe_error
#   └── xpipe_batch_error
#       ├── xpipe_batch_config        # invalid or incomplete xpipe_config object
#       ├── xpipe_batch_schema_domain # method/stream schema domain mismatch
#       ├── xpipe_batch_method        # unknown or invalid method identifier
#       ├── xpipe_batch_stream        # unknown stream identifier
#       ├── xpipe_batch_path          # inbox path invalid or not found
#       ├── xpipe_batch_input_type    # file vs directory requirement mismatch
#       ├── xpipe_batch_input_files    # directory input file missing or ambiguous
#       ├── xpipe_batch_id            # invalid explicit batch_id
#       ├── xpipe_batch_collision     # target batch directory already exists
#       ├── xpipe_batch_input         # input missing or empty at materialization
#       ├── xpipe_batch_instrument    # extractor not registered or not loaded
#       ├── xpipe_batch_extraction    # extractor returned invalid result
#       ├── xpipe_batch_materialization # ctx or config invalid at materialization
#       └── xpipe_batch_abort         # user declined confirmation prompt

#' Ingest a Data Batch
#'
#' High-level, irreversible operation that converts an inbox input
#' into an immutable Batch artifact.
#'
#' @description
#' \code{ingest_batch()} performs a complete batch ingestion.
#' The operation runs in three logical steps:
#' 1. Preparation (validation and resolution)
#' 2. Recap (human-readable intent summary)
#' 3. Materialization (irreversible)
#'
#' During preparation, configuration, input paths, and method/stream
#' compatibility are fully validated. During materialization, data is
#' extracted, privacy rules are applied, and the canonical batch
#' representation is persisted to disk.
#'
#' In interactive sessions, a recap and confirmation step is shown
#' after preparation and before materialization.
#'
#' @section Phases:
#' \itemize{
#'   \item \strong{Preparation}: validation and resolution only; no I/O
#'   \item \strong{Materialization}: extraction and persistence (irreversible)
#' }
#'
#' @section Immutability:
#' Once materialized, a batch is considered immutable. Re-ingestion of
#' the same input will create a new batch with a new batch identifier.
#'
#' @section Inspection:
#' After ingestion, batch contents and metadata can be inspected via
#' the Batch layer filesystem and inspection helpers.
#'
#' @param filename Character scalar. Name of a file or directory located in the inbox.
#' @param stream Character scalar. Target stream identifier.
#' @param method Character scalar. Ingestion method identifier.
#' @param config An \code{xpipe_config} object loaded via \code{load_options()}.
#' @param batch_id Optional explicit batch identifier.
#' @param confirm Logical; whether to ask for confirmation. Defaults to \code{interactive()}.
#' @param dry_run Logical; if TRUE, performs preparation and recap only.
#'
#' @return Character scalar. The generated batch identifier.
#'
#' @details
#' This function performs irreversible filesystem side effects.
#' It is recommended to validate inputs and configuration before use.
#'
#' @export
ingest_batch <- function(
  filename,
  stream,
  method,
  config = get_xpipe_config(),
  batch_id = NULL,
  confirm = interactive(),
  dry_run = FALSE
) {
  checkmate::assert_string(filename)
  checkmate::assert_string(stream)
  checkmate::assert_string(method)
  checkmate::assert_class(config, "xpipe_config")
  checkmate::assert_string(batch_id, null.ok = TRUE)
  checkmate::assert_flag(confirm)
  checkmate::assert_flag(dry_run)

  ctx <- .prepare_batch_ingestion(
    filename = filename,
    stream = stream,
    method = method,
    config = config,
    batch_id = batch_id
  )

  .print_ingestion_recap(ctx, action = "Batch ingestion")

  if (isTRUE(dry_run)) {
    cli::cli_alert_info(
      "Dry run complete. No files were written and no state was changed."
    )
    return(invisible(ctx$batch_id))
  }

  .confirm_ingestion(confirm = confirm)

  .materialize_batch(ctx, config)

  invisible(ctx$batch_id)
}

#' Prepare Batch Ingestion Context
#'
#' Constructs a validated batch ingestion context based on an input file
#' or directory located in the inbox and a configured ingestion method.
#'
#' This function performs configuration validation, input resolution,
#' and method/stream compatibility checks, but does not read, transform,
#' or write any data. The returned list represents a fully specified
#' ingestion intent ready for materialization.
#'
#' @param filename Character scalar. Name of a file or directory located
#'   in the inbox path defined in the local configuration.
#' @param stream Character scalar identifying the target stream.
#' @param method Character scalar identifying the ingestion method to use,
#'   as defined in \code{methods.json}.
#' @param config An object of class \code{xpipe_config}, as returned by
#'   \code{load_options()}.
#' @param batch_id Optional character scalar. Explicit batch identifier.
#'   If \code{NULL}, a unique batch identifier is generated automatically.
#'
#' @return A named list containing the ingestion context.
#'
#' @details
#' The context list encapsulates all metadata required for batch
#' materialization, including resolved input paths, method and stream
#' configuration, and target batch directory.
#'
#' No filesystem side effects are performed beyond path validation.
#' Data extraction and persistence occur only during
#' \code{materialize_batch()}.
#'
#' @keywords internal
.prepare_batch_ingestion <- function(
  filename,
  stream,
  method,
  config,
  batch_id = NULL
) {
  .assert_config_for_batch(config)

  method_config <- .get_method_config(config, method)
  stream_config <- .get_stream_config(config, stream)
  stream_id <- stream

  if (method_config$schema_domain != stream_config$schema_domain) {
    cli::cli_abort(
      c(
        "Invalid batch job",
        "x" = "Method and stream belong to different schema domains",
        "i" = "Method domain: {.val {method_config$schema_domain}}",
        "i" = "Stream domain: {.val {stream_config$schema_domain}}"
      ),
      class = c("xpipe_batch_schema_domain", "xpipe_batch_error", "xpipe_error")
    )
  }

  filepath <- .resolve_inbox_path(config, filename)

  # Validation against method requirement
  .validate_input_type(filepath, method_config)

  # Input files contract
  input_files <- method_config$input_files %||% NULL


  # Discover and resolve inputs
  resolved_inputs <- if (fs::is_dir(filepath)) {

    # method declares directory => input_files required
    if (identical(method_config$input_type, "directory") && is.null(input_files)) {
      cli::cli_abort(
        c(
          "Invalid method configuration",
          "x" = "Directory methods must define {.field input_files}"
        ),
        class = c("xpipe_batch_input_files", "xpipe_batch_error", "xpipe_error")
      )
    }

    .resolve_inputs(filepath, input_files)

  } else {

    # single-file input: do not replicate per component anymore
    list(data = filepath)

  }



  batch_id <- .normalize_or_generate_batch_id(stream_id, method, batch_id)
  batch_dir <- .build_batch_target_dir(config, batch_id)

  list(
    batch_id = batch_id,
    filename = filename,
    filepath = filepath,
    method_id = method,
    method_config = method_config,
    stream_id = stream_id,
    stream_config = stream_config,
    batch_dir = batch_dir,
    resolved_inputs = resolved_inputs
  )
}


## Init helpers --------------------------------------------------------

#' Assert configuration validity for batch creation
#'
#' Ensures that a valid \code{xpipe_config} object is provided and that
#' all required branches (local paths, streams, methods) are present.
#'
#' @param config An object of class \code{xpipe_config}.
#' @return Invisibly returns TRUE if valid, otherwise aborts.
#' @keywords internal
#' @noRd
.assert_config_for_batch <- function(config) {
  if (!inherits(config, "xpipe_config")) {
    cli::cli_abort(
      c(
        "Invalid configuration",
        "x" = "Expected an {.cls xpipe_config} object from {.code load_xpipe_config()}"
      ),
      class = c("xpipe_batch_config", "xpipe_batch_error", "xpipe_error")
    )
  }

  required_paths <- c("inbox", "batch")
  missing_paths <- setdiff(required_paths, names(config$local$paths))

  if (length(missing_paths) > 0) {
    cli::cli_abort(
      c(
        "Invalid configuration",
        "x" = "Missing required paths in config: {.val {missing_paths}}"
      ),
      class = c("xpipe_batch_config", "xpipe_batch_error", "xpipe_error")
    )
  }

  if (is.null(config$streams) || is.null(config$methods)) {
    cli::cli_abort(
      c(
        "Invalid configuration",
        "x" = "Streams or Methods options not loaded"
      ),
      class = c("xpipe_batch_config", "xpipe_batch_error", "xpipe_error")
    )
  }

  invisible(TRUE)
}


#' Retrieve method configuration
#'
#' @param config xpipe_config object.
#' @param method_id Character scalar.
#' @return A list with the method configuration.
#' @keywords internal
#' @noRd
.get_method_config <- function(config, method_id) {
  checkmate::assert_string(method_id)

  available <- names(config$methods)
  if (!(method_id %in% available)) {
    cli::cli_abort(
      c(
        "Invalid method",
        "x" = "Unknown method: {.val {method_id}}",
        "i" = "Available methods: {.str {available}}"
      ),
      class = c("xpipe_batch_method", "xpipe_batch_error", "xpipe_error")
    )
  }

  config$methods[[method_id]]
}

#' Retrieve stream configuration
#'
#' @param config xpipe_config object.
#' @param stream_id Character scalar.
#' @return A list with the stream configuration.
#' @keywords internal
#' @noRd
.get_stream_config <- function(config, stream_id) {
  checkmate::assert_string(stream_id)

  available <- names(config$streams)
  if (!(stream_id %in% available)) {
    cli::cli_abort(
      c(
        "Invalid stream",
        "x" = "Unknown stream: {.val {stream_id}}",
        "i" = "Available streams: {.str {available}}"
      ),
      class = c("xpipe_batch_stream", "xpipe_batch_error", "xpipe_error")
    )
  }

  config$streams[[stream_id]]
}


#' Resolve input path from inbox
#'
#' @param config xpipe_config object.
#' @param path Character scalar.
#' @return Normalized absolute path.
#' @keywords internal
#' @noRd
.resolve_inbox_path <- function(config, path) {
  checkmate::assert_string(path)

  full_path <- fs::path(config$local$paths$inbox, path)
  if (!fs::file_exists(full_path)) {
    cli::cli_abort(
      c(
        "Input path not found",
        "x" = "Path {.file {path}} was not found in inbox",
        "i" = "Inbox: {.file {config$local$paths$inbox}}"
      ),
      class = c("xpipe_batch_path", "xpipe_batch_error", "xpipe_error")
    )
  }

  fs::path_norm(full_path)
}

#' Validate input type against method requirement
#'
#' @keywords internal
#' @noRd
.validate_input_type <- function(filepath, method_config) {
  required_type <- method_config$input_type
  actual_is_dir <- fs::is_dir(filepath)

  if (required_type == "file" && actual_is_dir) {
    cli::cli_abort(
      c(
        "Invalid input type",
        "x" = "Selected method requires a file, but a directory was provided.",
        "i" = "Path: {.file {filepath}}"
      ),
      class = c("xpipe_batch_input_type", "xpipe_batch_error", "xpipe_error")
    )
  }

  if (required_type == "directory" && !actual_is_dir) {
    cli::cli_abort(
      c(
        "Invalid input type",
        "x" = "Method {.val {method_config$method_id}} requires a directory, but a file was provided.",
        "i" = "Path: {.file {filepath}}"
      ),
      class = c("xpipe_batch_input_type", "xpipe_batch_error", "xpipe_error")
    )
  }
}

#' Discover and resolve input_files
#'
#' Resolves logical input_files into concrete file paths.
#'
#' @keywords internal
#' @noRd
.resolve_inputs <- function(dir_path, input_files) {

  if (is.null(input_files)) return(NULL)

  resolved <- list()

  for (nm in names(input_files)) {
    pattern <- input_files[[nm]]
    matches <- fs::dir_ls(dir_path, glob = pattern)

    if (length(matches) == 0) {
      cli::cli_abort(
        c(
          "Missing input",
          "x" = "No files found for input {.val {nm}}",
          "i" = "Pattern: {.val {pattern}}",
          "i" = "Directory: {.file {dir_path}}"
        ),
        class = c("xpipe_batch_input_files", "xpipe_batch_error", "xpipe_error")
      )
    }

    if (length(matches) > 1) {
      cli::cli_abort(
        c(
          "Ambiguous input",
          "x" = "Multiple files found for input {.val {nm}}",
          "i" = "Pattern: {.val {pattern}}",
          "i" = "Found: {.file {basename(matches)}}"
        ),
        class = c("xpipe_batch_input_files", "xpipe_batch_error", "xpipe_error")
      )
    }

    resolved[[nm]] <- as.character(matches[[1]])
  }

  resolved
}


#' Normalize or Generate Batch Identifier
#'
#' Validates an explicit batch identifier or generates a unique
#' stream-scoped batch ID when none is provided.
#'
#' @param stream_id Character scalar identifying the stream.
#' @param method_id Character scalar identifying the method.
#' @param batch_id Optional character scalar. Explicit batch ID.
#'
#' @return Character scalar batch identifier.
#'
#' @noRd
.normalize_or_generate_batch_id <- function(
  stream_id,
  method_id,
  batch_id = NULL
) {
  checkmate::assert_string(stream_id)
  checkmate::assert_string(method_id)
  if (!is.null(batch_id)) {
    checkmate::assert_string(batch_id)
    if (!nzchar(batch_id)) {
      cli::cli_abort(
        c(
          "Invalid batch identifier",
          "x" = "{.arg batch_id} must be a non-empty string"
        ),
        class = c("xpipe_batch_id", "xpipe_batch_error", "xpipe_error")
      )
    }
    return(batch_id)
  }
  ts <- format(Sys.time(), "%Y%m%dT%H%M%OS3")
  paste0(ts, "_", stream_id, "_", method_id)
}

#' Build Batch Target Directory Path
#'
#' Constructs the normalized filesystem path for a batch within
#' the Batch layer, based on stream and batch identifiers.
#'
#' @param config xpipe_config object.
#' @param batch_id Character scalar batch identifier.
#'
#' @return Normalized filesystem path to the batch directory.
#'
#' @noRd
.build_batch_target_dir <- function(config, batch_id) {
  checkmate::assert_string(batch_id)
  fs::path_norm(fs::path(config$local$paths$batch, batch_id))
}


#' Print ingestion recap
#'
#' Prints a standardized recap of an ingestion intent before
#' an irreversible operation is performed.
#'
#' @param ctx Ingestion context list.
#' @param action Character scalar describing the action (e.g. "Batch ingestion").
#'
#' @keywords internal
#' @noRd
.print_ingestion_recap <- function(ctx, action = "Ingestion") {
  cli::cli_h2("{action} recap")

  cli::cli_text(
    "The following input will be processed and materialized as an immutable artifact."
  )

  cli::cli_h3("Input")
  cli::cli_bullets(c(
    "Path   : {.file {ctx$filepath}}",
    "Stream : {.val {ctx$stream_id}}",
    "Method : {.val {ctx$method_id}}"
  ))

  if (!is.null(ctx$resolved_inputs)) {
    cli::cli_h3("Resolved input files")
    cli::cli_bullets(
      purrr::imap_chr(
        ctx$resolved_inputs,
        ~ paste0("{.strong ", .y, "} : {.file ", basename(.x), "}")
      )
    )
  }

  cli::cli_h3("Target")
  cli::cli_bullets(c(
    "Batch ID        : {.val {ctx$batch_id}}",
    "Batch directory : {.file {ctx$batch_dir}}"
  ))

  cli::cli_alert_warning(
    "This operation is irreversible once materialization starts."
  )

  invisible(TRUE)
}

#' Confirm irreversible ingestion
#'
#' @param confirm Logical; whether confirmation is required.
#'
#' @return Invisibly TRUE if confirmed, otherwise aborts.
#'
#' @keywords internal
#' @noRd
.confirm_ingestion <- function(confirm = TRUE) {
  if (!isTRUE(confirm)) {
    return(invisible(TRUE))
  }

  # Try climenu first, fall back cleanly if unsupported
  if (requireNamespace("climenu", quietly = TRUE)) {
    choice <- tryCatch(
      climenu::select(
        c("Yes, continue", "No, abort"),
        prompt = "Proceed with materialization?"
      ),
      error = function(e) NULL
    )

    if (!is.null(choice)) {
      if (choice != "Yes, continue") {
        cli::cli_abort(
          "Batch ingestion aborted by user.",
          class = c("xpipe_batch_abort", "xpipe_batch_error", "xpipe_error")
        )
      }
      return(invisible(TRUE))
    }
  }

  # Fallback: readline (always works)
  answer <- readline("Proceed with materialization? [y/N]: ")

  if (!nzchar(answer) || tolower(substr(answer, 1, 1)) != "y") {
    cli::cli_abort(
      "Batch ingestion aborted by user.",
      class = c("xpipe_batch_abort", "xpipe_batch_error", "xpipe_error")
    )
  }

  invisible(TRUE)
}

# Batch materialization pipeline ------------------------------------------
#
# Conceptual phases:
#   Phase A – Pre-materialization guardrails
#   Phase B – Extraction instrument resolution
#   Phase C – Data extraction (in-memory)
#   Phase D – Batch persistence (canonical output)
#
# Only Phase A is implemented as a standalone conceptual function.
# Phases B-D are sequential pipeline steps used exclusively by
# `materialize_batch()`.


#' Enforce method-declared column types on extracted Batch tables
#'
#' Applies hard type enforcement according to method_config$column_types.
#' Aborts if coercion would cause semantic loss or is not possible.
#'
#' @param extracted_data Named list of data.frames
#' @param method_config Method configuration list
#'
#' @return Modified extracted_data
#' @keywords internal
.apply_batch_column_types <- function(extracted_data, method_config) {

  column_types <- method_config$column_types %||% NULL
  if (is.null(column_types)) {
    return(extracted_data)
  }

  for (table_name in names(column_types)) {

    if (!(table_name %in% names(extracted_data))) {
      cli::cli_warn(
        "Declared column_types for table {.val {table_name}}, but table was not extracted."
      )
      next
    }

    df <- extracted_data[[table_name]]
    spec <- column_types[[table_name]]

    for (col in names(spec)) {

      target_type <- spec[[col]]

      if (!(col %in% names(df))) {
        cli::cli_warn(
          "Declared column type for {.val {table_name}.{col}}, but column is absent."
        )
        next
      }

      x <- df[[col]]

      if (target_type == "character") {

        # If already character, ok
        if (is.character(x)) {
          next
        }

        # If numeric or integer, coercion would lose leading zeros → abort
        if (is.numeric(x)) {
          cli::cli_abort(
            c(
              "Batch type enforcement failed",
              "x" = "Column {.val {table_name}.{col}} is numeric but declared as character",
              "i" = "This may indicate loss of semantic information (e.g. leading zeros)",
              "i" = "Fix the extractor or upstream parsing logic"
            ),
            class = c(
              "xpipe_batch_type_enforcement",
              "xpipe_batch_materialization",
              "xpipe_batch_error",
              "xpipe_error"
            )
          )
        }

        # Fallback: attempt safe as.character
        df[[col]] <- as.character(x)
        next
      }

      # Unsupported target types (for now)
      cli::cli_abort(
        c(
          "Batch type enforcement failed",
          "x" = "Unsupported target type {.val {target_type}} for {.val {table_name}.{col}}"
        ),
        class = c(
          "xpipe_batch_type_enforcement",
          "xpipe_batch_materialization",
          "xpipe_batch_error",
          "xpipe_error"
        )
      )
    }

    extracted_data[[table_name]] <- df
  }

  extracted_data
}


#' Materialize Batch Ingestion Context
#'
#' Transforms a validated ingestion context into an immutable, on-disk
#' batch representation.
#'
#' @param ctx Ingestion context list.
#' @param config xpipe_config object.
#' @return Invisibly TRUE on success.
#' @keywords internal
.materialize_batch <- function(ctx, config) {
  .assert_batch_materializable(ctx, config)

  instrument <- .resolve_extraction_instrument(ctx, config)

  cli::cli_alert_info("Running extraction")

  extracted_data <- rlang::try_fetch(
    .run_extraction(ctx, instrument),
    error = function(e) {
      cli::cli_abort(
        "Data extraction failed for method type {.val {ctx$method_config$type}}",
        parent = e,
        class = c("xpipe_batch_extraction", "xpipe_batch_error", "xpipe_error")
      )
    }
  )

  cli::cli_alert_info("Applying privacy rules")

  for (nm in names(extracted_data)) {
    extracted_data[[nm]] <- apply_privacy_rules(
      df = extracted_data[[nm]],
      table_name = nm,
      method_config = ctx$method_config,
      pepper = config$runtime$pepper
    )
  }

  cli::cli_alert_info("Enforcing batch column types")

  extracted_data <- .apply_batch_column_types(
    extracted_data = extracted_data,
    method_config  = ctx$method_config
  )


  rlang::try_fetch(
    .persist_to_batch(ctx, extracted_data, config),
    error = function(e) {
      cli::cli_abort(
        "Failed to persist batch data to {.file {ctx$batch_dir}}",
        parent = e,
        class = c(
          "xpipe_batch_materialization",
          "xpipe_batch_error",
          "xpipe_error"
        )
      )
    }
  )

  invisible(TRUE)
}


## Phases ----------------------------------------------------------------

#' Assert that batch materialization is safe
#'
#' Performs Phase A (Materialization Guardrails) checks.
#' Validates that the ingestion context is coherent and that
#' filesystem preconditions are met before any side effects occur.
#'
#' @param ctx Ingestion context list.
#' @param config xpipe_config object.
#' @return Invisibly TRUE if materializable, otherwise aborts.
#' @keywords internal
#' @noRd
.assert_batch_materializable <- function(ctx, config) {
  checkmate::assert_list(ctx, min.len = 1)
  checkmate::assert_class(config, "xpipe_config")

  # A1. Target collision check (batch layer)
  if (fs::dir_exists(ctx$batch_dir)) {
    cli::cli_abort(
      c(
        "Ingestion cannot proceed",
        "x" = "Target batch directory already exists",
        "i" = "Path: {.file {ctx$batch_dir}}",
        "i" = "Batch id: {.val {ctx$batch_id}}"
      ),
      class = c("xpipe_batch_collision", "xpipe_batch_error", "xpipe_error")
    )
  }

  # A2. Input path re-validation
  if (!fs::file_exists(ctx$filepath)) {
    cli::cli_abort(
      c(
        "Ingestion cannot proceed",
        "x" = "Input path does not exist at materialization time",
        "i" = "Expected path: {.file {ctx$filepath}}"
      ),
      class = c("xpipe_batch_input", "xpipe_batch_error", "xpipe_error")
    )
  }

  # Lightweight sanity check
  info <- fs::file_info(ctx$filepath)
  if (is.na(info$size) || (fs::is_file(ctx$filepath) && info$size == 0)) {
    cli::cli_abort(
      c(
        "Ingestion cannot proceed",
        "x" = "Input is empty or unreadable",
        "i" = "Path: {.file {ctx$filepath}}"
      ),
      class = c("xpipe_batch_input", "xpipe_batch_error", "xpipe_error")
    )
  }

  invisible(TRUE)
}

#' Resolve extraction instrument
#'
#' @keywords internal
#' @noRd
.resolve_extraction_instrument <- function(ctx, config) {
  # Registry: method type -> extractor function name (string)
  # Add new extractors here; the function must be loaded before ingest_batch() is called.
  .instrument_registry <- c(
    "access_mdb" = "extract_access_db",
    "xml" = "extract_xml"
  )

  method_type <- ctx$method_config$type

  fn_name <- .instrument_registry[method_type]

  if (is.na(fn_name)) {
    cli::cli_abort(
      c(
        "No extraction instrument registered",
        "x" = "No extractor registered for method type {.val {method_type}}",
        "i" = "Registered types: {.val {names(.instrument_registry)}}"
      ),
      class = c("xpipe_batch_instrument", "xpipe_batch_error", "xpipe_error")
    )
  }

  if (!exists(fn_name, mode = "function")) {
    cli::cli_abort(
      c(
        "Extraction instrument not loaded",
        "x" = "Function {.code {fn_name}} is not available in the current environment",
        "i" = "Source the appropriate tool script before calling {.code ingest_batch()}"
      ),
      class = c("xpipe_batch_instrument", "xpipe_batch_error", "xpipe_error")
    )
  }

  get(fn_name, mode = "function")
}


#' Run extraction and load batch data into memory
#'
#' @keywords internal
#' @noRd
.run_extraction <- function(ctx, instrument) {
  extracted <- instrument(ctx)

  if (is.data.frame(extracted)) {
    extracted <- list(data = extracted)
  }

  if (!is.list(extracted) || length(extracted) == 0) {
    cli::cli_abort(
      c(
        "Invalid extraction result",
        "x" = "Extractor must return a non-empty list of data frames"
      ),
      class = c("xpipe_batch_extraction", "xpipe_batch_error", "xpipe_error")
    )
  }


  if (is.null(names(extracted)) || any(!nzchar(names(extracted)))) {
    cli::cli_abort(
      c(
        "Invalid extraction result",
        "x" = "Extracted list must have a valid name for every element"
      ),
      class = c("xpipe_batch_extraction", "xpipe_batch_error", "xpipe_error")
    )
  }

  invalid_elements <- purrr::keep(extracted, ~ !is.data.frame(.x))
  if (length(invalid_elements) > 0) {
    cli::cli_abort(
      c(
        "Invalid extracted object",
        "x" = "All extracted elements must be data.frames",
        "i" = "Invalid element(s): {.val {names(invalid_elements)}}"
      ),
      class = c("xpipe_batch_extraction", "xpipe_batch_error", "xpipe_error")
    )
  }

  extracted
}


#' Persist extracted data to the Batch layer
#'
#' @keywords internal
#' @noRd
.persist_to_batch <- function(ctx, extracted_data, config) {
  fs::dir_create(ctx$batch_dir, recurse = TRUE)

  anchor <- ctx$method_config$anchor_table
  written_tables <- character(0)

  for (nm in names(extracted_data)) {
    tbl <- extracted_data[[nm]]

    # zero-column guard
    if (!is.data.frame(tbl) || ncol(tbl) == 0) {
      if (identical(nm, anchor)) {
        cli::cli_abort(
          c(
            "Invalid batch artifact",
            "x" = "Anchor table {.val {nm}} has zero columns and cannot be materialized"
          ),
          class = c("xpipe_batch_materialization", "xpipe_batch_error", "xpipe_error")
        )
      }

      cli::cli_alert_warning(
        "Not writing {.val {nm}}: extracted table has zero columns."
      )
      next
    }

    arrow::write_parquet(
      tbl,
      fs::path(ctx$batch_dir, paste0(nm, ".parquet"))
    )

    written_tables <- c(written_tables, nm)
  }

  # anchor must have been materialized
  if (!(anchor %in% written_tables)) {
    cli::cli_abort(
      c(
        "Invalid batch artifact",
        "x" = "Anchor table {.val {anchor}} was not materialized"
      ),
      class = c("xpipe_batch_materialization", "xpipe_batch_error", "xpipe_error")
    )
  }

  # Meta file as JSON for cross-platform compatibility
  meta <- list(
    batch_id = ctx$batch_id,

    stream = list(
      id = ctx$stream_id,
      schema_domain = ctx$method_config$schema_domain,
      anchor = ctx$stream_config$anchor
    ),

    method = list(
      id = ctx$method_id,
      type = ctx$method_config$type
    ),

    input = list(
      path = as.character(ctx$filepath),
      input_type = ctx$method_config$input_type
    ),

    execution = list(
      created_at = format(Sys.time(), "%Y-%m-%dT%H:%M:%S%z"),
      tool = config$runtime$tool$name,
      tool_version = config$runtime$tool$version
    )
  )

  if (!is.null(ctx$resolved_inputs)) {
    meta$resolved_inputs <- lapply(ctx$resolved_inputs, basename)
  }

  meta$tables <- written_tables

  jsonlite::write_json(
    meta,
    fs::path(ctx$batch_dir, "_batch_metadata.json"),
    pretty = TRUE,
    auto_unbox = TRUE
  )

  cli::cli_alert_success("Batch materialized: {.val {ctx$batch_id}}")

  invisible(TRUE)
}

#       ├── xpipe_bronze_anchor      # anchor table/column/key definition or join failure

#' Promote materialized batches to the Bronze layer
#'
#' @description
#' Promotes immutable Batch artifacts into the Bronze layer as an
#' append-only, historical ledger.
#'
#' Each promotion:
#' \itemize{
#'   \item Appends rows only (no deletes, updates, or rewrites)
#'   \item Derives domain time from a single anchor table per batch
#'   \item Enriches all rows with mandatory Bronze metadata
#'   \item Partitions data by anchor-derived time (year or month)
#' }
#'
#' Bronze promotion preserves full history and lineage and does not
#' attempt deduplication, identity resolution, or "current state"
#' determination. These responsibilities are explicitly deferred to
#' the Silver layer.
#'
#' @section Preview and safety:
#' By default, Bronze promotion prints a preview showing:
#' \itemize{
#'   \item Batches to be promoted
#'   \item Anchor source used for time derivation
#'   \item Affected partition years
#'   \item Tables that will be appended
#' }
#'
#' Use \code{dry_run = TRUE} to perform validation and preview only,
#' without writing any data and without updating Bronze state.
#'
#' @section Storage layout:
#' Bronze is written as immutable Parquet shards per promoted batch:
#' \itemize{
#'   \item \code{bronze/<stream>__<method>/year=YYYY/<table>__<batch_id>.parquet}
#' }
#' Promotion refuses to overwrite an existing shard file.
#'
#' @section Anchor inheritance:
#' Anchor time is derived from the method anchor table only. Secondary tables
#' inherit \code{__anchor_time} by joining to the anchor table using
#' \code{identity_keys}. If required keys are missing or if rows cannot be matched,
#' promotion aborts for that batch.
#'
#' @section Immutability:
#' Bronze promotion is irreversible. Once rows are appended (written as shards),
#' they are never modified or removed.
#'
#' @param config xacc_config object.
#' @param stream Optional stream filter. If provided, only batches
#'   for the specified stream are considered.
#' @param dry_run Logical. If TRUE, performs validation and preview
#'   only. No files are written and no state is updated.
#' @param confirm Logical. If TRUE (default in interactive sessions),
#'   asks for confirmation before committing changes.
#'
#' @return Invisibly TRUE on success.
#'
#' @export
promote_to_bronze <- function(
    config = get_xpipe_config(),
    stream = NULL,
    dry_run = FALSE,
    confirm = interactive()
) {
  checkmate::assert_class(config, "xpipe_config")
  checkmate::assert_string(stream, null.ok = TRUE)
  checkmate::assert_flag(dry_run)
  checkmate::assert_flag(confirm)

  batch_root <- config$local$paths$batch
  checkmate::assert_string(batch_root)

  batch_dirs <- fs::dir_ls(batch_root, type = "directory", recurse = FALSE)
  if (length(batch_dirs) == 0) return(invisible(TRUE))

  # filtra batch validi (con metadata) e, se richiesto, per stream
  candidates <- character(0)
  metas <- list()

  for (bd in batch_dirs) {
    meta_path <- fs::path(bd, "_batch_metadata.json")
    if (!fs::file_exists(meta_path)) next

    meta <- rlang::try_fetch(jsonlite::fromJSON(meta_path, simplifyVector = FALSE),
                             error = function(e) NULL)
    if (is.null(meta)) next
    if (!is.null(stream) && !identical(meta$stream$id, stream)) next

    candidates <- c(candidates, bd)
    metas[[bd]] <- meta
  }

  if (length(candidates) == 0) return(invisible(TRUE))

  if (isTRUE(dry_run)) {
    for (bd in candidates) {
      cli::cli_inform("Would promote batch {.val {metas[[bd]]$batch_id}}")
    }
    return(invisible(TRUE))
  }

  # conferma una sola volta
  .confirm_bronze_promotion(confirm)

  for (bd in candidates) {
    meta <- metas[[bd]]

    tryCatch(
      .promote_batch_to_bronze(batch_dir = bd, config = config),
      xpipe_bronze_append_only = function(e) {
        cli::cli_inform("Skipping already promoted batch {.val {meta$batch_id}}")
        invisible(NULL)
      }
    )
  }

  invisible(TRUE)
}


# Helpers -----------------------------------------------------------------
#' Execute a statement and always clear the DBI result
#' @keywords internal
.db_exec <- function(con, sql, params = NULL) {
  res <- NULL
  on.exit({
    if (!is.null(res)) try(DBI::dbClearResult(res), silent = TRUE)
  }, add = TRUE)

  res <- DBI::dbSendStatement(con, sql, params = params)
  invisible(DBI::dbGetRowsAffected(res))
}

#' Fetch a query and always clear the DBI result
#' @keywords internal
.db_query <- function(con, sql, params = NULL) {
  res <- NULL
  on.exit({
    if (!is.null(res)) try(DBI::dbClearResult(res), silent = TRUE)
  }, add = TRUE)

  res <- DBI::dbSendQuery(con, sql, params = params)
  DBI::dbFetch(res)
}

#' Fetch a single scalar (first row/col) and clear the result
#' @keywords internal
.db_scalar <- function(con, sql, params = NULL) {
  out <- .db_query(con, sql, params = params)
  if (nrow(out) == 0 || ncol(out) == 0) return(NA)
  out[[1]][[1]]
}

#' Return TRUE if a Parquet file has at least one column
#' @keywords internal
.parquet_has_columns <- function(con, parquet_path) {
  pq_lit <- DBI::dbQuoteLiteral(con, as.character(parquet_path))
  n_cols <- .db_scalar(con,
                       paste0("SELECT COUNT(*) FROM parquet_schema(", pq_lit, ")")
  )
  isTRUE(!is.na(n_cols) && n_cols > 0)
}

#' Reconcile Bronze table columns with incoming Parquet schema
#'
#' Adds new source columns found in the incoming Parquet file to an existing
#' Bronze table. Existing rows receive NULL for the new columns.
#'
#' This helper only handles additive schema evolution. Incompatible type changes
#' are intentionally not handled here.
#'
#' @param con DBI connection.
#' @param table_name Character scalar. Bronze table name.
#' @param parquet_path Character scalar. Path to incoming Parquet file.
#'
#' @return Invisibly TRUE.
#' @keywords internal
.reconcile_bronze_columns_from_parquet <- function(con, table_name, parquet_path) {
  checkmate::assert_class(con, "DBIConnection")
  checkmate::assert_string(table_name)
  checkmate::assert_string(parquet_path)

  pq_lit <- DBI::dbQuoteLiteral(con, as.character(parquet_path))

  incoming_schema <- .db_query(
    con,
    paste0("DESCRIBE SELECT * FROM read_parquet(", pq_lit, ")")
  )

  incoming_cols <- incoming_schema$column_name
  existing_cols <- DBI::dbListFields(con, table_name)

  new_cols <- setdiff(incoming_cols, existing_cols)

  if (length(new_cols) == 0) {
    return(invisible(TRUE))
  }

  for (col in new_cols) {
    dtype <- incoming_schema$column_type[incoming_schema$column_name == col][[1]]

    cli::cli_warn(
      "Schema evolution: adding column {.val {col}} ({dtype}) to {.val {table_name}}"
    )

    .db_exec(
      con,
      paste0(
        "ALTER TABLE ",
        DBI::dbQuoteIdentifier(con, table_name),
        " ADD COLUMN ",
        DBI::dbQuoteIdentifier(con, col),
        " ",
        dtype
      )
    )
  }

  invisible(TRUE)
}

#' Open a DuckDB connection for the Bronze layer
#'
#' Opens (or creates, if missing) the DuckDB database file associated
#' with a given stream in the Bronze layer.
#'
#' The database file is located at:
#'   {bronze_root}/{stream_id}.duckdb
#'
#' This function does not create tables and does not perform any writes
#' by itself. It only establishes a DBI connection.
#'
#' DuckDB runs in-process. The returned connection points to a single
#' physical file on disk and must be explicitly closed by the caller
#' using DBI::dbDisconnect().
#'
#' @param bronze_root Character scalar. Path to the Bronze layer root directory.
#' @param stream_id Character scalar. Stream identifier (used as file name).
#' @param read_only Logical. If TRUE, opens the database in read-only mode.
#'
#' @return A DBI connection to the DuckDB database for the given stream.
#'
#' @examples
#' \dontrun{
#' con <- .open_bronze_conn("data/bronze", "sdo")
#' DBI::dbDisconnect(con, shutdown = TRUE)
#' }
#'
#' @keywords internal
.open_bronze_conn <- function(bronze_root, stream_id, read_only = FALSE) {
  checkmate::assert_string(bronze_root)
  checkmate::assert_string(stream_id)
  checkmate::assert_flag(read_only)

  db_path <- fs::path(bronze_root, paste0(stream_id, ".duckdb"))

  DBI::dbConnect(
    duckdb::duckdb(),
    dbdir = as.character(db_path),
    read_only = read_only
  )
}


#' Ensure Bronze promotion log table exists
#'
#' Creates the authoritative Bronze promotion log table
#' if it does not already exist.
#'
#' The promotion log records only successful promotions.
#' A batch is considered promoted if and only if its batch_id
#' appears in this table.
#'
#' This function is idempotent and safe to call multiple times.
#' It performs no inserts and does not start or end transactions.
#'
#' @param con A live DBI connection to a Bronze DuckDB database.
#'
#' @return Invisibly TRUE.
#'
#' @keywords internal
.ensure_bronze_promotion_log <- function(con) {
  checkmate::assert_class(con, "DBIConnection")

  sql <- "
    CREATE TABLE IF NOT EXISTS __promotion_log (
      batch_id         TEXT PRIMARY KEY,
      stream_id        TEXT NOT NULL,
      method_id        TEXT NOT NULL,
      promoted_at      TIMESTAMP NOT NULL,
      batch_created_at TIMESTAMP,
      anchor_source    TEXT,
      tables           TEXT,
      tool             TEXT,
      tool_version     TEXT
    )
  "

  DBI::dbExecute(con, sql)

  invisible(TRUE)
}




#' Promote one Batch table into Bronze (DuckDB)
#'
#' Appends a single Parquet table from a materialized Batch directory into the
#' Bronze DuckDB database for the corresponding stream, using an atomic
#' transaction and an authoritative promotion log.
#'
#' The batch is considered promoted if and only if its `batch_id` is recorded
#' in `__promotion_log`.
#'
#' @param batch_dir Character scalar. Path to a Batch directory.
#' @param source_table Character scalar. Table name inside the batch
#'   (expects `<source_table>.parquet`).
#' @param config xpipe_config. Used to locate `paths$bronze`.
#'
#' @return Invisibly TRUE on success; aborts on error.
#' @keywords internal
.promote_one_table_to_bronze <- function(batch_dir, source_table, config) {
  checkmate::assert_string(batch_dir)
  checkmate::assert_string(source_table)
  checkmate::assert_class(config, "xpipe_config")

  meta_file <- "_batch_metadata.json"
  meta_path <- fs::path(batch_dir, meta_file)
  if (!fs::file_exists(meta_path)) {
    cli::cli_abort(
      c("Batch metadata not found",
        "x" = "Missing file: {.val {meta_file}} in {.file {batch_dir}}"),
      class = c("xpipe_batch_metadata", "xpipe_batch_error", "xpipe_error")
    )
  }

  meta <- rlang::try_fetch(
    jsonlite::fromJSON(meta_path, simplifyVector = FALSE),
    error = function(e) cli::cli_abort(
      c("Invalid batch metadata JSON",
        "x" = "Failed to parse {.val {meta_file}} in {.file {batch_dir}}"),
      parent = e,
      class = c("xpipe_batch_metadata", "xpipe_batch_error", "xpipe_error")
    )
  )

  batch_id         <- meta$batch_id %||% NA_character_
  stream_id        <- meta$stream$id %||% NA_character_
  method_id        <- meta$method$id %||% NA_character_
  batch_created_at <- meta$execution$created_at %||% NA_character_
  tool             <- meta$execution$tool %||% NA_character_
  tool_version     <- meta$execution$tool_version %||% NA_character_

  if (!checkmate::test_string(batch_id, min.chars = 1) ||
      !checkmate::test_string(stream_id, min.chars = 1) ||
      !checkmate::test_string(method_id, min.chars = 1) ||
      !checkmate::test_string(batch_created_at, min.chars = 1)) {
    cli::cli_abort(
      c("Batch metadata missing required fields",
        "x" = "Required: batch_id, stream.id, method.id, execution.created_at",
        "i" = "Batch metadata file in {.file {meta_path}}"),
      class = c("xpipe_batch_metadata", "xpipe_batch_error", "xpipe_error")
    )
  }

  parquet_path <- fs::path(batch_dir, paste0(source_table, ".parquet"))
  if (!fs::file_exists(parquet_path)) {
    cli::cli_abort(
      c("Batch table parquet not found",
        "x" = "Missing file: {.val {source_table}}.parquet in {.path {batch_dir}}"),
      class = c("xpipe_batch_materialization", "xpipe_batch_error", "xpipe_error")
    )
  }

  bronze_root <- config$local$paths$bronze
  if (!checkmate::test_string(bronze_root, min.chars = 1)) {
    cli::cli_abort(
      c("Bronze path not configured",
        "x" = "Expected config$local$paths$bronze to be set"),
      class = c("xpipe_bronze_state", "xpipe_bronze_error", "xpipe_error")
    )
  }

  con <- .open_bronze_conn(bronze_root, stream_id, read_only = FALSE)
  on.exit(DBI::dbDisconnect(con, shutdown = TRUE), add = TRUE)

  .ensure_bronze_promotion_log(con)

  bronze_table <- paste0(stream_id, "__", method_id, "__", source_table)
  tbl_ident <- DBI::dbQuoteIdentifier(con, bronze_table)
  pq_lit <- DBI::dbQuoteLiteral(con, as.character(parquet_path))

  promoted_at <- format(Sys.time(), "%Y-%m-%d %H:%M:%S")

  DBI::dbBegin(con)

  rlang::try_fetch(
    {
      exists_in_log <- DBI::dbGetQuery(
        con,
        "SELECT COUNT(*) AS n FROM __promotion_log WHERE batch_id = ?",
        params = list(batch_id)
      )$n[[1]] > 0

      if (isTRUE(exists_in_log)) {
        cli::cli_abort(
          c("Append-only violation",
            "x" = "Batch {.val {batch_id}} is already recorded in __promotion_log"),
          class = c("xpipe_bronze_append_only", "xpipe_bronze_error", "xpipe_error")
        )
      }

      DBI::dbExecute(
        con,
        paste0(
          "CREATE TABLE IF NOT EXISTS ", tbl_ident, " AS
           SELECT
             t.*,
             CAST(NULL AS TEXT)      AS __batch_id,
             CAST(NULL AS TEXT)      AS __stream_id,
             CAST(NULL AS TEXT)      AS __method_id,
             CAST(NULL AS TIMESTAMP) AS __batch_created_at,
             CAST(NULL AS TIMESTAMP) AS __bronze_promoted_at
           FROM read_parquet(", pq_lit, ") t
           WHERE 1=0"
        )
      )

      DBI::dbExecute(
        con,
        paste0(
          "INSERT INTO ", tbl_ident, " BY NAME
           SELECT
             t.*,
             ? AS __batch_id,
             ? AS __stream_id,
             ? AS __method_id,
             CAST(? AS TIMESTAMP) AS __batch_created_at,
             CAST(? AS TIMESTAMP) AS __bronze_promoted_at
           FROM read_parquet(", pq_lit, ") t"
        ),
        params = list(batch_id, stream_id, method_id, batch_created_at, promoted_at)
      )

      DBI::dbExecute(
        con,
        "INSERT INTO __promotion_log
         (batch_id, stream_id, method_id, promoted_at, batch_created_at,
          anchor_source, tables, tool, tool_version)
         VALUES (?, ?, ?, CAST(? AS TIMESTAMP), CAST(? AS TIMESTAMP), ?, ?, ?, ?)",
        params = list(
          batch_id, stream_id, method_id,
          promoted_at, batch_created_at,
          NA_character_,
          jsonlite::toJSON(bronze_table, auto_unbox = TRUE),
          tool, tool_version
        )
      )

      DBI::dbCommit(con)
      invisible(TRUE)
    },
    error = function(e) {
      DBI::dbRollback(con)
      if (inherits(e, "xpipe_bronze_append_only")) stop(e)
      cli::cli_abort(
        c("Bronze promotion failed and was rolled back",
          "x" = "No partial data should remain for batch {.val {batch_id}}"),
        parent = e,
        class = c("xpipe_bronze_error", "xpipe_error")
      )
    }
  )
}


#' Promote an entire Batch into Bronze (DuckDB)
#'
#' Promotes all Parquet tables in a materialized Batch directory into the
#' Bronze DuckDB database for the corresponding stream, using a single
#' atomic transaction and a single authoritative promotion log entry.
#'
#' A batch is considered promoted if and only if its `batch_id` is present
#' in `__promotion_log`.
#'
#' @param batch_dir Character scalar. Path to a Batch directory.
#' @param config xpipe_config. Used to locate `paths$bronze`.
#'
#' @return Invisibly TRUE on success; aborts on error.
#' @keywords internal
.promote_batch_to_bronze <- function(batch_dir, config) {
  checkmate::assert_string(batch_dir)
  checkmate::assert_class(config, "xpipe_config")

  meta_file <- "_batch_metadata.json"
  meta_path <- fs::path(batch_dir, meta_file)
  if (!fs::file_exists(meta_path)) {
    cli::cli_abort(
      c("Batch metadata not found",
        "x" = "Missing file: {.val {meta_file}} in {.file {batch_dir}}"),
      class = c("xpipe_batch_metadata", "xpipe_batch_error", "xpipe_error")
    )
  }

  meta <- rlang::try_fetch(
    jsonlite::fromJSON(meta_path, simplifyVector = FALSE),
    error = function(e) cli::cli_abort(
      c("Invalid batch metadata JSON",
        "x" = "Failed to parse {.val {meta_file}} in {.file {batch_dir}}"),
      parent = e,
      class = c("xpipe_batch_metadata", "xpipe_batch_error", "xpipe_error")
    )
  )

  batch_id         <- meta$batch_id %||% NA_character_
  stream_id        <- meta$stream$id %||% NA_character_
  method_id        <- meta$method$id %||% NA_character_
  batch_created_at <- meta$execution$created_at %||% NA_character_
  tool             <- meta$execution$tool %||% NA_character_
  tool_version     <- meta$execution$tool_version %||% NA_character_

  if (!checkmate::test_string(batch_id, min.chars = 1) ||
      !checkmate::test_string(stream_id, min.chars = 1) ||
      !checkmate::test_string(method_id, min.chars = 1) ||
      !checkmate::test_string(batch_created_at, min.chars = 1)) {
    cli::cli_abort(
      c("Batch metadata missing required fields",
        "x" = "Required: batch_id, stream.id, method.id, execution.created_at",
        "i" = "Batch metadata file in {.file {meta_path}}"),
      class = c("xpipe_batch_metadata", "xpipe_batch_error", "xpipe_error")
    )
  }


  method_cfg <- config$methods[[method_id]]

  if (is.null(method_cfg)) {
    cli::cli_abort(
      c(
        "Unknown method in batch metadata",
        "x" = "Method {.val {method_id}} not found in methods contract"
      ),
      class = c("xpipe_methods_error", "xpipe_error")
    )
  }

  anchor_tbl <- method_cfg$anchor_table %||% NA_character_

  if (!checkmate::test_string(anchor_tbl, min.chars = 1)) {
    cli::cli_abort(
      c(
        "Invalid method configuration",
        "x" = "Method {.val {method_id}} must define anchor_table"
      ),
      class = c("xpipe_bronze_anchor", "xpipe_bronze_error", "xpipe_error")
    )
  }


  parquet_files <- fs::dir_ls(batch_dir, glob = "*.parquet")
  if (length(parquet_files) == 0) {
    cli::cli_abort(
      c("No Parquet tables found in batch",
        "x" = "Expected one or more *.parquet files in {.file {batch_dir}}"),
      class = c("xpipe_batch_materialization", "xpipe_batch_error", "xpipe_error")
    )
  }

  source_tables <- fs::path_ext_remove(fs::path_file(parquet_files))
  # anchor table must exist in batch
  if (!(anchor_tbl %in% source_tables)) {
    cli::cli_abort(
      c(
        "Invalid batch for Bronze promotion",
        "x" = "Anchor table {.val {anchor_tbl}} is missing in batch {.val {batch_id}}"
      ),
      class = c("xpipe_bronze_anchor", "xpipe_bronze_error", "xpipe_error")
    )
  }


  bronze_root <- config$local$paths$bronze
  if (!checkmate::test_string(bronze_root, min.chars = 1)) {
    cli::cli_abort(
      c("Bronze path not configured",
        "x" = "Expected config$local$paths$bronze to be set"),
      class = c("xpipe_bronze_state", "xpipe_bronze_error", "xpipe_error")
    )
  }

  con <- .open_bronze_conn(bronze_root, stream_id, read_only = FALSE)
  on.exit(DBI::dbDisconnect(con, shutdown = TRUE), add = TRUE)

  .ensure_bronze_promotion_log(con)

  promoted_at <- format(Sys.time(), "%Y-%m-%d %H:%M:%S")
  promoted_tables <- character(0)

  DBI::dbBegin(con)

  rlang::try_fetch(
    {
      exists_in_log <- .db_scalar(
        con,
        "SELECT COUNT(*) AS n FROM __promotion_log WHERE batch_id = ?",
        params = list(batch_id)
      ) > 0

      if (isTRUE(exists_in_log)) {
        cli::cli_abort(
          c("Append-only violation",
            "x" = "Batch {.val {batch_id}} is already recorded in __promotion_log"),
          class = c("xpipe_bronze_append_only", "xpipe_bronze_error", "xpipe_error")
        )
      }

      for (tbl in source_tables) {
        parquet_path <- fs::path(batch_dir, paste0(tbl, ".parquet"))

        if (!fs::file_exists(parquet_path)) {
          cli::cli_warn("Skipping missing parquet for table {.val {tbl}} in batch {.val {batch_id}}")
          next
        }

        if (!.parquet_has_columns(con, parquet_path)) {
          cli::cli_warn("Skipping {.file {basename(parquet_path)}}: Parquet has zero columns.")
          next
        }

        pq_lit <- DBI::dbQuoteLiteral(con, as.character(parquet_path))

        bronze_table <- paste0(stream_id, "__", method_id, "__", tbl)
        tbl_ident <- DBI::dbQuoteIdentifier(con, bronze_table)

        .db_exec(
          con,
          paste0(
            "CREATE TABLE IF NOT EXISTS ", tbl_ident, " AS
           SELECT
             t.*,
             CAST(NULL AS TEXT)      AS __batch_id,
             CAST(NULL AS TEXT)      AS __stream_id,
             CAST(NULL AS TEXT)      AS __method_id,
             CAST(NULL AS TIMESTAMP) AS __batch_created_at,
             CAST(NULL AS TIMESTAMP) AS __bronze_promoted_at
           FROM read_parquet(", pq_lit, ") t
           WHERE 1=0"
          )
        )

        .reconcile_bronze_columns_from_parquet(
          con = con,
          table_name = bronze_table,
          parquet_path = parquet_path
        )

        .db_exec(
          con,
          paste0(
            "INSERT INTO ", tbl_ident, " BY NAME
           SELECT
             t.*,
             ? AS __batch_id,
             ? AS __stream_id,
             ? AS __method_id,
             CAST(? AS TIMESTAMP) AS __batch_created_at,
             CAST(? AS TIMESTAMP) AS __bronze_promoted_at
           FROM read_parquet(", pq_lit, ") t"
          ),
          params = list(batch_id, stream_id, method_id, batch_created_at, promoted_at)
        )

        promoted_tables <- c(promoted_tables, bronze_table)
      }

      .db_exec(
        con,
        "INSERT INTO __promotion_log
       (batch_id, stream_id, method_id, promoted_at, batch_created_at,
        anchor_source, tables, tool, tool_version)
       VALUES (?, ?, ?, CAST(? AS TIMESTAMP), CAST(? AS TIMESTAMP), ?, ?, ?, ?)",
        params = list(
          batch_id, stream_id, method_id,
          promoted_at, batch_created_at,
          NA_character_,
          jsonlite::toJSON(promoted_tables, auto_unbox = TRUE),
          tool, tool_version
        )
      )

      DBI::dbCommit(con)
      invisible(TRUE)
    },
    error = function(e) {
      try(.db_exec(con, "ROLLBACK"), silent = TRUE)
      if (inherits(e, "xpipe_bronze_append_only")) stop(e)
      cli::cli_abort(
        c("Bronze batch promotion failed",
          "x" = "Promotion did not complete; no partial state should remain."),
        parent = e,
        class = c("xpipe_bronze_error", "xpipe_error")
      )
    }
  )
}

# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
# 2. silver_promotion.R
#
# Promotes Bronze data to the Silver layer.
#
# Approach:
# Builds Silver-current tables from Bronze DuckDB tables. Currentness is defined
# by the latest batch containing each entity, ordered by __batch_created_at and
# __batch_id. Secondary tables are filtered by joining to the current anchor
# table using identity keys plus __batch_id.
#
# Public API: promote_to_silver()
# Internals: .promote_to_silver_current(), .silver_exec(), .silver_query(),
#            .resolve_silver_canonical_method(), .is_silver_up_to_date(),
#            .confirm_silver_promotion()
# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~


# Error Class Taxonomy ----------------------------------------------------



# Public API --------------------------------------------------------------

#' Promote data to Silver
#'
#' Uses silver.json to resolve canonical methods and build Silver-current tables.
#' Harmonized phase is intentionally not implemented in this branch.
#'
#' @param config xpipe_config.
#' @param stream Optional stream id. If NULL, promotes all enabled streams in silver.json.
#' @param phase Character. Only "current" is supported.
#' @param dry_run Logical. If TRUE, prints what would be done.
#' @param confirm Logical. Ask for confirmation before writing.
#'
#' @return Invisibly TRUE.
#' @export
promote_to_silver <- function(
    config = get_xpipe_config(),
    stream = NULL,
    phase = "current",
    dry_run = FALSE,
    confirm = interactive()
) {
  checkmate::assert_class(config, "xpipe_config")
  checkmate::assert_string(stream, null.ok = TRUE)
  checkmate::assert_string(phase)
  checkmate::assert_flag(dry_run)
  checkmate::assert_flag(confirm)

  if (is.null(config$silver) || length(config$silver) == 0) {
    cli::cli_abort(
      c(
        "Silver contract not loaded",
        "x" = "No silver streams found in config. Ensure data/shared/silver.json is present and valid."
      ),
      class = c("xpipe_config_error", "xpipe_error")
    )
  }

  if (!identical(phase, "current")) {
    cli::cli_abort(
      c(
        "Unsupported Silver phase",
        "x" = "Only {.val {'current'}} is supported in this branch."
      ),
      class = c("xpipe_silver_error", "xpipe_error")
    )
  }

  targets <- .select_silver_promotion_targets(
    config = config,
    stream = stream
  )

  if (length(targets) == 0) {
    cli::cli_inform("No Silver-current streams enabled.")
    return(invisible(TRUE))
  }

  if (isTRUE(dry_run)) {
    for (sid in targets) {
      mid <- .resolve_silver_canonical_method(sid, config)
      cli::cli_inform(
        "Would promote Silver-current for stream {.val {sid}} using method {.val {mid}}."
      )
    }
    return(invisible(TRUE))
  }

  .confirm_silver_promotion(confirm)

  for (sid in targets) {
    .promote_to_silver_current(
      stream_id = sid,
      config = config
    )
  }

  invisible(TRUE)
}

# Helpers -----------------------------------------------------------------


#' Execute a Silver SQL statement and always clear the result
#'
#' @param con DBI connection.
#' @param sql SQL statement.
#' @param params Optional query parameters.
#'
#' @return Invisibly returns rows affected.
#' @keywords internal
.silver_exec <- function(con, sql, params = NULL) {
  res <- NULL
  on.exit({
    if (!is.null(res)) try(DBI::dbClearResult(res), silent = TRUE)
  }, add = TRUE)

  res <- DBI::dbSendStatement(con, sql, params = params)
  invisible(DBI::dbGetRowsAffected(res))
}

#' Run a Silver SQL query and always clear the result
#'
#' @param con DBI connection.
#' @param sql SQL query.
#' @param params Optional query parameters.
#'
#' @return A data.frame.
#' @keywords internal
.silver_query <- function(con, sql, params = NULL) {
  res <- NULL
  on.exit({
    if (!is.null(res)) try(DBI::dbClearResult(res), silent = TRUE)
  }, add = TRUE)

  res <- DBI::dbSendQuery(con, sql, params = params)
  DBI::dbFetch(res)
}


.select_silver_promotion_targets <- function(config, stream = NULL) {
  checkmate::assert_class(config, "xpipe_config")
  checkmate::assert_string(stream, null.ok = TRUE)

  silver_streams <- names(config$silver)

  if (length(silver_streams) == 0) {
    cli::cli_abort(
      c(
        "Silver contract not loaded",
        "x" = "No streams found in silver.json."
      ),
      class = c("xpipe_config_error", "xpipe_error")
    )
  }

  enabled <- silver_streams[vapply(
    silver_streams,
    function(sid) isTRUE(config$silver[[sid]]$current$enabled),
    logical(1)
  )]

  if (!is.null(stream)) {
    if (!(stream %in% silver_streams)) {
      cli::cli_abort(
        c(
          "Invalid Silver stream",
          "x" = "Stream {.val {stream}} not found in silver.json"
        ),
        class = c("xpipe_config_error", "xpipe_error")
      )
    }

    if (!isTRUE(config$silver[[stream]]$current$enabled)) {
      cli::cli_abort(
        c(
          "Silver-current disabled",
          "x" = "Silver-current is disabled for stream {.val {stream}}"
        ),
        class = c("xpipe_config_error", "xpipe_error")
      )
    }

    return(stream)
  }

  if (length(enabled) == 0) {
    return(character(0))
  }

  # Non-interactive: preserve pipeline behavior.
  # stream = NULL means all enabled streams.
  if (!interactive()) {
    return(enabled)
  }

  choices <- c("All enabled streams", enabled, "Abort")

  choice <- NULL
  if (requireNamespace("climenu", quietly = TRUE)) {
    choice <- tryCatch(
      climenu::select(
        choices,
        prompt = "Select Silver stream to promote"
      ),
      error = function(e) NULL
    )
  }

  if (is.null(choice)) {
    cat("\nSelect Silver stream to promote\n\n")
    for (i in seq_along(choices)) {
      cat(sprintf(" %d. %s\n", i, choices[[i]]))
    }
    cat("\n")

    k <- suppressWarnings(as.integer(readline("Enter number: ")))

    if (is.na(k) || k < 1 || k > length(choices)) {
      cli::cli_abort(
        "Invalid selection.",
        class = c("xpipe_silver_abort", "xpipe_silver_error", "xpipe_error")
      )
    }

    choice <- choices[[k]]
  }

  if (identical(choice, "Abort")) {
    cli::cli_abort(
      "Silver promotion aborted by user.",
      class = c("xpipe_silver_abort", "xpipe_silver_error", "xpipe_error")
    )
  }

  if (identical(choice, "All enabled streams")) {
    return(enabled)
  }

  choice
}

# Resolve canonical method for Silver-current
# @noRd
.resolve_silver_canonical_method <- function(stream_id, config) {
  if (is.null(config$silver[[stream_id]])) {
    cli::cli_abort(
      c(
        "Silver not configured for stream",
        "x" = "Stream {.val {stream_id}} not found in silver.json"
      ),
      class = c("xpipe_config_error", "xpipe_error")
    )
  }

  s <- config$silver[[stream_id]]

  if (!isTRUE(s$current$enabled)) {
    cli::cli_abort(
      c(
        "Silver-current disabled",
        "x" = "Silver-current is disabled for stream {.val {stream_id}}"
      ),
      class = c("xpipe_config_error", "xpipe_error")
    )
  }

  s$canonical_method
}

.is_silver_up_to_date <- function(con, stream_id, method_id, anchor_tbl) {

  latest_bronze <- .silver_query(
    con,
    paste0(
      "SELECT MAX(batch_id) AS batch_id
       FROM bronze.main.__promotion_log
       WHERE stream_id = ",
      DBI::dbQuoteLiteral(con, stream_id),
      " AND method_id = ",
      DBI::dbQuoteLiteral(con, method_id)
    )
  )$batch_id[[1]]

  current_anchor <- paste0("current__", anchor_tbl)

  silver_latest <- if (current_anchor %in% DBI::dbListTables(con)) {
    .silver_query(
      con,
      paste0(
        "SELECT MAX(__batch_id) AS batch_id
         FROM ", DBI::dbQuoteIdentifier(con, current_anchor)
      )
    )$batch_id[[1]]
  } else {
    NA_character_
  }


  if (is.na(latest_bronze) || is.na(silver_latest)) {
    return(FALSE)
  }

  identical(latest_bronze, silver_latest)

}

## Pipeline ----------------------------------------------------------------


#' Promote Bronze data to Silver-current (DuckDB)
#'
#' Builds Silver-current tables for a given stream and method by selecting the
#' latest batch per entity from the anchor table and collecting secondary tables
#' via `(identity_keys + __batch_id)`.
#'
#' Currentness is defined by ingestion order:
#' `__batch_created_at DESC, __batch_id DESC`.
#'
#' @param stream_id Character scalar. Stream identifier.
#' @param method_id Character scalar. Method identifier.
#' @param config xpipe_config.
#'
#' @return Invisibly TRUE on success.
#' @keywords internal
.promote_to_silver_current <- function(stream_id, config) {

  # 1. Preconditions

  checkmate::assert_string(stream_id)
  checkmate::assert_class(config, "xpipe_config")

  method_id <- .resolve_silver_canonical_method(stream_id, config)

  bronze_root <- config$local$paths$bronze
  silver_root <- config$local$paths$silver

  if (!checkmate::test_string(bronze_root, min.chars = 1) ||
      !checkmate::test_string(silver_root, min.chars = 1)) {
    cli::cli_abort(
      c(
        "Silver promotion failed",
        "x" = "Both paths$bronze and paths$silver must be configured"
      ),
      class = c("xpipe_silver_state", "xpipe_error")
    )
  }

  method_cfg <- config$methods[[method_id]]
  if (is.null(method_cfg)) {
    cli::cli_abort(
      c(
        "Invalid Silver configuration",
        "x" = "Canonical method {.val {method_id}} not found in methods contract"
      ),
      class = c("xpipe_methods_error", "xpipe_error")
    )
  }

  anchor_tbl <- method_cfg$anchor_table %||% NA_character_
  keys <- method_cfg$identity_keys %||% character(0)

  if (!checkmate::test_string(anchor_tbl, min.chars = 1) ||
      !checkmate::test_character(keys, min.len = 1)) {
    cli::cli_abort(
      c(
        "Invalid Silver configuration",
        "x" = "Method must define anchor_table and identity_keys"
      ),
      class = c("xpipe_silver_config", "xpipe_error")
    )
  }

  bronze_db <- fs::path(bronze_root, paste0(stream_id, ".duckdb"))
  silver_db <- fs::path(silver_root, paste0(stream_id, ".duckdb"))

  if (!fs::file_exists(bronze_db)) {
    cli::cli_abort(
      c(
        "Bronze database not found",
        "x" = "Expected {.file {bronze_db}} to exist"
      ),
      class = c("xpipe_bronze_state", "xpipe_error")
    )
  }

  fs::dir_create(silver_root, recurse = TRUE)

  # 2. Open Silver and attach Bronze

  con_s <- DBI::dbConnect(
    duckdb::duckdb(),
    dbdir = as.character(silver_db),
    read_only = FALSE
  )

  on.exit({
    try(.silver_exec(con_s, "DETACH bronze"), silent = TRUE)
    DBI::dbDisconnect(con_s, shutdown = TRUE)
  }, add = TRUE)

  try(.silver_exec(con_s, "DETACH bronze"), silent = TRUE)

  .silver_exec(
    con_s,
    paste0(
      "ATTACH ",
      DBI::dbQuoteLiteral(con_s, as.character(bronze_db)),
      " AS bronze (READ_ONLY)"
    )
  )

  # 3. Discover Bronze tables

  prefix <- paste0(stream_id, "__", method_id, "__")
  anchor_full <- paste0(prefix, anchor_tbl)

  bronze_tables <- .silver_query(
    con_s,
    "SELECT table_name
     FROM duckdb_tables()
     WHERE database_name = 'bronze'
       AND schema_name = 'main'"
  )$table_name

  method_tables <- bronze_tables[startsWith(bronze_tables, prefix)]

  if (!(anchor_full %in% method_tables)) {
    cli::cli_abort(
      c(
        "Anchor table not found in Bronze",
        "x" = "Expected {.val {anchor_full}} to exist"
      ),
      class = c("xpipe_silver_anchor", "xpipe_error")
    )
  }

  current_anchor <- paste0("current__", anchor_tbl)

  key_cols <- DBI::dbQuoteIdentifier(con_s, keys)
  keys_sql <- paste(key_cols, collapse = ", ")

  bronze_anchor <- paste0(
    "bronze.main.",
    DBI::dbQuoteIdentifier(con_s, anchor_full)
  )

  # 4. Up-to-date check

  if (.is_silver_up_to_date(con_s, stream_id, method_id, anchor_tbl)) {

    if (!interactive()) {
      cli::cli_inform(
        "Silver-current for stream {.val {stream_id}} left unchanged (already up to date)."
      )
      return(invisible(TRUE))
    }

    answer <- readline(
      paste0(
        "Silver-current is already up to date for stream ",
        stream_id, ". Rebuild anyway? [y/N]: "
      )
    )

    if (!nzchar(answer) || tolower(substr(answer, 1, 1)) != "y") {
      cli::cli_inform(
        "Silver-current for stream {.val {stream_id}} left unchanged."
      )
      return(invisible(TRUE))
    }

    cli::cli_inform(
      "Rebuilding Silver-current for stream {.val {stream_id}} despite being up to date."
    )
  }

  # 5. Transactional rebuild

  DBI::dbBegin(con_s)

  rlang::try_fetch(
    {
      .silver_exec(
        con_s,
        paste0(
          "CREATE OR REPLACE TABLE ",
          DBI::dbQuoteIdentifier(con_s, current_anchor), " AS
           SELECT * EXCLUDE (rn)
           FROM (
             SELECT
               a.*,
               ROW_NUMBER() OVER (
                 PARTITION BY ", keys_sql, "
                 ORDER BY __batch_created_at DESC, __batch_id DESC
               ) AS rn
             FROM ", bronze_anchor, " a
           ) t
           WHERE rn = 1"
        )
      )

      for (bt in method_tables) {
        short <- sub(prefix, "", bt, fixed = TRUE)
        if (identical(short, anchor_tbl)) next

        bronze_sec <- paste0(
          "bronze.main.",
          DBI::dbQuoteIdentifier(con_s, bt)
        )

        current_sec <- paste0("current__", short)

        join_keys <- paste(
          paste0(
            "s.", DBI::dbQuoteIdentifier(con_s, keys),
            " = a.", DBI::dbQuoteIdentifier(con_s, keys)
          ),
          collapse = " AND "
        )

        .silver_exec(
          con_s,
          paste0(
            "CREATE OR REPLACE TABLE ",
            DBI::dbQuoteIdentifier(con_s, current_sec), " AS
             SELECT s.*
             FROM ", bronze_sec, " s
             INNER JOIN ", DBI::dbQuoteIdentifier(con_s, current_anchor), " a
               ON ", join_keys, "
              AND s.__batch_id = a.__batch_id"
          )
        )
      }

      DBI::dbCommit(con_s)

      cli::cli_alert_success(
        "Silver-current rebuilt for stream {.val {stream_id}} using method {.val {method_id}}."
      )

      invisible(TRUE)
    },
    error = function(e) {
      try(DBI::dbRollback(con_s), silent = TRUE)
      cli::cli_abort(
        c(
          "Silver-current promotion failed",
          "x" = "Silver tables were not rebuilt successfully."
        ),
        parent = e,
        class = c("xpipe_silver_error", "xpipe_error")
      )
    }
  )
}

#' Confirm Silver promotion
#'
#' @param confirm Logical.
#' @keywords internal
#' @noRd
.confirm_silver_promotion <- function(confirm = interactive()) {
  checkmate::assert_flag(confirm)

  if (!isTRUE(confirm)) return(invisible(TRUE))

  if (requireNamespace("climenu", quietly = TRUE) && interactive()) {

      ),
      error = function(e) NULL
    )
    if (!is.null(choice)) {
      if (choice != "Yes, continue") {
        cli::cli_abort(
          "Silver promotion aborted by user.",
          class = c("xpipe_silver_abort", "xpipe_silver_error", "xpipe_error")
        )
      }
      return(invisible(TRUE))
    }
  }

  answer <- readline("Proceed with Silver promotion? [y/N]: ")
  if (!nzchar(answer) || tolower(substr(answer, 1, 1)) != "y") {
    cli::cli_abort(
      "Silver promotion aborted by user.",
      class = c("xpipe_silver_abort", "xpipe_silver_error", "xpipe_error")
    )
  }

  invisible(TRUE)
}

'''
R/core_config.R
'''
# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
# core_config.R
#
# Configuration and options loading for the xpipe control plane.
#
# Approach: Parses and validates JSON configurations and schema files,
# manages runtime context (like the pepper key), and normalizes paths.
# It ensures that local infrastructure and remote layer configurations
# are properly structured before any pipeline tasks occur.
#
# Public API:  load_xpipe_config()
# Internals:   .find_asset(), .validate_json_schema(), .read_json_safe(),
#              .load_runtime_context(), .load_local_config(), .normalize_paths(),
#              .validate_batch_layer(), .validate_batch_hash(), .load_batch_layer(),
#              .load_streams_contract(), .load_methods_contract()
# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~


# Internal Environment ----------------------------------------------------
#
# This is to hold package state
.xpipe_env <- new.env(parent = emptyenv())

# Error Class Taxonomy --------------------------------------------------------
#
# All errors in this file inherit from `xpipe_error` (package base class),
# allowing callers to catch any xpipe error at a single level if needed.
#
# Hierarchy (specific → domain → base):
#
#   xpipe_error                        # catch-all for the package
#   ├── xpipe_runtime_error
#   │   └── xpipe_runtime_pepper       # XNE_PEPPER_KEY missing or empty
#   ├── xpipe_config_error
#   │   ├── xpipe_config_missing       # load_xpipe_config() not yet called
#   │   └── xpipe_config_schema_domain # method references unknown stream domain
#   ├── xpipe_local_error
#   │   └── xpipe_local_paths          # missing or non-existent path in local.json
#   ├── xpipe_batch_error
#   │   ├── xpipe_batch_paths          # batch directory missing or not configured
#   │   ├── xpipe_batch_metadata       # layer.json missing or invalid
#   │   └── xpipe_batch_hash           # HMAC integrity check failed
#   ├── xpipe_streams_error
#   │   ├── xpipe_streams_metadata     # _meta block missing or invalid type
#   │   └── xpipe_streams_definition   # stream missing fields or invalid anchor
#   ├── xpipe_methods_error            # methods contract load or schema failure
#   ├── xpipe_json_error               # (composed into domain errors via error_class)
#   │   ├── xpipe_json_not_found       # file does not exist
#   │   ├── xpipe_json_schema          # file fails schema validation
#   │   └── xpipe_json_invalid         # file is not valid JSON
#   └── xpipe_asset_error
#       └── xpipe_asset_not_found      # internal inst/ asset not locatable
#
# Usage in tryCatch / try_fetch:
#   rlang::try_fetch(expr, xpipe_error = function(e) ...)   # catch anything
#   rlang::try_fetch(expr, xpipe_batch_error = function(e) ...) # catch batch only

# Public API ---------------------------------------------------------------

#' Load and Validate Configuration
#'
#' Loads the full xpipe control-plane configuration, including:
#' - runtime context
#' - local infrastructure configuration
#' - batch layer metadata
#' - streams contract (domain semantics)
#' - methods contract (ingestion mechanics)
#'
#' @param path Path to local.json.
#' @param project_root Project root.
#' @return xpipe_config object.
#' @export
load_xpipe_config <- function(path = NULL, project_root = NULL) {
  checkmate::assert_string(path, null.ok = TRUE)
  checkmate::assert_string(project_root, null.ok = TRUE)

  config <- list()

  config$runtime <- .config_step("initialize runtime context",
                                 .load_runtime_context()
  )
  config$local   <- .config_step("load local configuration",
                                 .load_local_config(path, project_root)
  )
  config$batch   <- .config_step("load Batch layer",
                                 .load_batch_layer(config$local, config$runtime)
  )

  streams_contract <- .config_step("load Streams contract",
                                   .load_streams_contract(config$local)
  )
  config$streams_meta <- streams_contract$meta
  config$streams      <- streams_contract$streams

  config$methods <- .config_step("load Methods contract",
                                 .load_methods_contract(config$local)
  )

  silver_contract <- .config_step(
    "load Silver contract",
    .load_silver_contract(config$local)
  )

  config$silver_meta <- silver_contract$meta
  config$silver      <- silver_contract$streams

  # Validate schema domains consistency
  stream_domains  <- unique(vapply(config$streams, `[[`, "", "schema_domain"))
  method_domains  <- unique(vapply(config$methods, `[[`, "", "schema_domain"))

  unknown_domains <- setdiff(method_domains, stream_domains)

  if (length(unknown_domains) > 0) {
    cli::cli_abort(
      c(
        "Invalid configuration",
        "x" = "Methods reference unknown schema_domain(s): {.val {unknown_domains}}",
        "i" = "Defined stream domains: {.val {stream_domains}}"
      ),
      class = c("xpipe_config_schema_domain", "xpipe_config_error", "xpipe_error")
    )
  }

  # Validate Silver canonical methods
  for (stream_id in names(config$silver)) {
    s <- config$silver[[stream_id]]

    if (isTRUE(s$current$enabled)) {
      method_id <- s$canonical_method

      if (!method_id %in% names(config$methods)) {
        cli::cli_abort(
          c(
            "Invalid silver contract",
            "x" = "Canonical method {.val {method_id}} for stream {.val {stream_id}} does not exist"
          ),
          class = c("xpipe_config_error", "xpipe_error")
        )
      }

      method_domain <- config$methods[[method_id]]$schema_domain
      stream_domain <- config$streams[[stream_id]]$schema_domain

      if (!identical(method_domain, stream_domain)) {
        cli::cli_abort(
          c(
            "Invalid silver contract",
            "x" = "Canonical method {.val {method_id}} domain {.val {method_domain}} does not match stream {.val {stream_id}} domain {.val {stream_domain}}"
          ),
          class = c("xpipe_config_error", "xpipe_error")
        )
      }
    }
  }

  cli::cli_alert_success("Configuration system loaded and validated.")

  config_obj <- structure(config, class = "xpipe_config")
  .xpipe_env$config <- config_obj

  invisible(config_obj)
}


# Internal API ------------------------------------------------------------


#' Retrieve Active xpipe Configuration
#'
#' @return An xpipe_config object.
#' @keywords internal
get_xpipe_config <- function() {
  if (is.null(.xpipe_env$config)) {
    cli::cli_abort(
      c(
        "Configuration not loaded",
        "x" = "No active configuration found in the package environment.",
        "i" = "Call {.code load_xpipe_config()} before running pipeline tasks."
      ),
      class = c("xpipe_config_missing", "xpipe_config_error", "xpipe_error")
    )
  }
  .xpipe_env$config
}



# Main Helpers (steps) --------------------------------------------------------

#' Load Runtime Context
#'
#' Initializes the ephemeral runtime context for the current process.
#'
#' The runtime context contains environment-bound and non-persistent
#' information such as cryptographic keys and load timestamps.
#'
#' This is NOT a data layer and does not represent persisted state.
#'
#' @return Runtime context configuration.
#' @noRd
.load_runtime_context <- function() {
  runtime <- list(
    tool = list(
      name    = "xpipe",
      version =
        if (requireNamespace("xpipe", quietly = TRUE)) {
          as.character(utils::packageVersion("xpipe"))
        } else {
          "dev"
        }
    ),
    pepper = Sys.getenv("XNE_PEPPER_KEY", unset = NA_character_),
    loaded_at = Sys.time()
  )

  if (!checkmate::test_string(runtime$pepper, min.chars = 1)) {
    cli::cli_abort(
      "Environment variable {.envvar XNE_PEPPER_KEY} not found.",
      class = c("xpipe_runtime_pepper", "xpipe_runtime_error", "xpipe_error")
    )
  }

  runtime
}

#' Load Local Configuration
#'
#' Loads and normalizes environment-specific infrastructure configuration.
#'
#' The local configuration defines filesystem paths, logging locations,
#' and other deployment-specific settings required to run the system.
#'
#' This is NOT a data layer and does not encode domain or semantic rules.
#'
#' @param path Path to local.json.
#' @param project_root Root for resolving relative paths.
#' @return Normalized local configuration.
#' @noRd
.load_local_config <- function(path = NULL, project_root = NULL) {
  if (is.null(path)) {
    path <- .find_asset("_xne", "local.json")
  }

  .validate_json_schema(
    json_path = path,
    schema_name = "local.schema.json",
    error_title = "Invalid local.json",
    error_class = "xpipe_local_error"
  )

  local_cfg <- .read_json_safe(
    path,
    error_title = "Invalid local configuration",
    error_class = "xpipe_local_error"
  )

  .normalize_paths(local_cfg, project_root)
}

#' Load Batch Layer Metadata
#'
#' Loads and validates the persistent metadata describing the Batch
#' medallion layer. This metadata represents STATE of the data layer,
#' not configuration or domain logic.
#'
#' @param local_config Local configuration.
#' @param runtime_config Runtime context.
#' @return Parsed and validated Batch layer metadata.
#' @noRd
.load_batch_layer <- function(local_config, runtime_config) {
  layer_json <- .validate_batch_layer(local_config)

  layer_options <- .read_json_safe(
    path = layer_json,
    error_title = "Invalid Batch layer metadata",
    error_class = "xpipe_batch_metadata"
  )

  .validate_batch_hash(layer_options, runtime_config$pepper)

  layer_options
}
#' Load Streams Contract
#'
#' Loads and validates the Streams contract, which defines domain-level
#' semantic invariants (streams) used across Batch, Bronze, and Silver.
#'
#' The Streams contract is GOVERNANCE METADATA, not configuration and not
#' a medallion data layer.
#'
#' @param local_config Local configuration.
#' @return A list with two elements:
#'   - meta: streams contract metadata
#'   - streams: named list of stream definitions
#' @noRd
.load_streams_contract <- function(local_config) {
  path <- fs::path(local_config$paths$shared, "streams.json")

  .validate_json_schema(
    json_path  = path,
    schema_name = "streams.schema.json",
    error_title = "Invalid streams.json",
    error_class = "xpipe_streams_error"
  )

  raw <- .read_json_safe(
    path,
    error_title = "Invalid streams contract",
    error_class = "xpipe_streams_error"
  )

  # Extract and validate contract metadata
  meta <- raw$`_meta`

  if (is.null(meta) || is.null(meta$type) || meta$type != "streams_contract") {
    cli::cli_abort(
      c(
        "Invalid streams contract",
        "x" = "Missing or invalid {.field _meta.type}",
        "i" = "Expected: 'streams_contract'"
      ),
      class = c("xpipe_streams_metadata", "xpipe_streams_error", "xpipe_error")
    )
  }

  # Extract actual streams
  streams <- raw[names(raw) != "_meta"]

  if (length(streams) == 0) {
    cli::cli_abort(
      c(
        "Invalid streams contract",
        "x" = "No stream definitions found"
      ),
      class = c("xpipe_streams_definition", "xpipe_streams_error", "xpipe_error")
    )
  }

  #  Minimal semantic validation
  for (nm in names(streams)) {
    s <- streams[[nm]]

    missing_fields <- setdiff(
      c("label", "anchor", "partition_granularity", "valid_from"),
      names(s)
    )

    if (length(missing_fields) > 0) {
      cli::cli_abort(
        c(
          "Invalid stream definition",
          "x" = "Stream {.val {nm}} missing required fields: {.val {missing_fields}}"
        ),
        class = c("xpipe_streams_definition", "xpipe_streams_error", "xpipe_error")
      )
    }

    if (!(s$anchor %in% c("start", "end"))) {
      cli::cli_abort(
        c(
          "Invalid stream definition",
          "x" = "Stream {.val {nm}} has invalid anchor: {.val {s$anchor}}",
          "i" = "Allowed: 'start', 'end'"
        ),
        class = c("xpipe_streams_definition", "xpipe_streams_error", "xpipe_error")
      )
    }
  }

  list(
    meta    = meta,
    streams = streams
  )
}

#' Load Methods Contract
#'
#' Loads and validates the Methods contract, which defines ingestion
#' mechanics and extraction rules used to materialize Batch data.
#'
#' A Method describes HOW data is ingested (format, input_files, source_tables,
#' partitioning, privacy), not WHAT the data means.
#'
#' The Methods contract is a CONTROL-PLANE artifact and must not
#' be confused with medallion data layers.
#'
#' @param local_config Local configuration.
#' @return Validated methods configuration.
#' @noRd
.load_methods_contract <- function(local_config) {
  path <- fs::path(local_config$paths$shared, "methods.json")

  .validate_json_schema(
    json_path = path,
    schema_name = "methods.schema.json",
    error_title = "Invalid methods.json",
    error_class = "xpipe_methods_error"
  )

  .read_json_safe(
    path,
    error_title = "Invalid methods contract",
    error_class = "xpipe_methods_error"
  )
}

#' Load Silver Contract
#'
#' Loads and validates the Silver contract, which defines the canonical
#' method used to build Silver-current per stream.
#'
#' @param local_config Local configuration.
#' @return Parsed silver contract.
#' @noRd
.load_silver_contract <- function(local_config) {
  path <- fs::path(local_config$paths$shared, "silver.json")

  .validate_json_schema(
    json_path   = path,
    schema_name = "silver_contract.schema.json",
    error_title = "Invalid silver.json",
    error_class = "xpipe_config_error"
  )

  raw <- .read_json_safe(
    path,
    error_title = "Invalid silver contract",
    error_class = "xpipe_config_error"
  )

  meta <- raw$`_meta`
  if (is.null(meta) || meta$type != "silver_contract") {
    cli::cli_abort(
      c(
        "Invalid silver contract",
        "x" = "Missing or invalid {.field _meta.type}",
        "i" = "Expected: 'silver_contract'"
      ),
      class = c("xpipe_config_error", "xpipe_error")
    )
  }

  streams <- raw$streams
  if (is.null(streams) || length(streams) == 0) {
    cli::cli_abort(
      c(
        "Invalid silver contract",
        "x" = "No streams defined"
      ),
      class = c("xpipe_config_error", "xpipe_error")
    )
  }

  list(
    meta    = meta,
    streams = streams
  )
}




# Sub-helpers --------------------------------------------------------

#' Wrap a Configuration Boot Step with Contextual Error Handling
#'
#' Executes a single configuration loading step and catches any
#' `xpipe_error` condition, rethrowing it as `xpipe_config_error`
#' with a step-level label prepended to the message chain.
#'
#' The original error is preserved as the parent condition and
#' rendered by cli as a "Caused by" line, so no diagnostic
#' information is lost.
#'
#' @param label A string describing the step, used in the outer
#'   error message as "Failed to {label}". Should complete the
#'   sentence naturally, e.g. `"load Batch layer"`.
#' @param expr A single expression to evaluate — typically one
#'   internal helper call. Eagerly evaluated, so multi-line
#'   blocks must be wrapped in `{}`.
#'
#' @return The result of `expr` if it succeeds.
#'
#' @section Error handling:
#'   Catches: any condition inheriting from `xpipe_error`.
#'   Throws:  `c("xpipe_config_error", "xpipe_error")`.
#'   Intermediate class handlers (e.g. `xpipe_batch_error`) are
#'   not needed here because all throws in this codebase use
#'   leaf-level classes, all of which inherit from `xpipe_error`.
#'
#' @noRd
.config_step <- function(label, expr) {
  rlang::try_fetch(
    expr,
    xpipe_error = function(e) cli::cli_abort(
      "Failed to {label}",
      parent = e,
      call = NULL,
      class = c("xpipe_config_error", "xpipe_error")
    )
  )
}

#' Find Internal Asset
#'
#' Resilient helper to locate internal files (config, schemas) in both
#' development and installed package modes.
#'
#' @param ... Character strings defining the subpath within 'inst'.
#' @return Character scalar. The absolute path to the asset.
#' @noRd
.find_asset <- function(...) {
  search_paths <- c(".", "..", "../..")

  for (base in search_paths) {
    dev_path <- fs::path_abs(fs::path(base, "inst", ...))
    if (fs::file_exists(dev_path) || fs::dir_exists(dev_path)) {
      return(dev_path)
    }
  }

  pkg_path <- system.file(..., package = "xpipe")
  if (pkg_path != "") {
    return(pkg_path)
  }

  cli::cli_abort(
    c(
      "Internal asset not found",
      "x" = "Could not locate {.file {fs::path(...)}}"
    ),
    class = c("xpipe_asset_not_found", "xpipe_asset_error", "xpipe_error")
  )
}

#' Validate JSON File Against Internal Schema
#'
#' @param json_path Path to JSON file.
#' @param schema_name Schema file name.
#' @param title Error title.
#' @param class Error classes.
#' @return Invisibly TRUE if valid.
#' @noRd
.validate_json_schema <- function(json_path, schema_name, error_title, error_class) {
  checkmate::assert_string(json_path)
  checkmate::assert_string(schema_name)
  checkmate::assert_string(error_title)
  checkmate::assert_string(error_class)


  if (!fs::file_exists(json_path)) {
    cli::cli_abort(
      c(error_title, "x" = "JSON file not found: {.file {json_path}}"),
      class = c(error_class, "xpipe_json_not_found", "xpipe_json_error", "xpipe_error")
    )
  }

  schema_path <- .find_asset("schema", schema_name)

  ok <- jsonvalidate::json_validate(
    json_path,
    schema = schema_path,
    engine = "ajv",
    verbose = TRUE
  )

  if (!ok) {
    cli::cli_abort(
      c(
        error_title,
        "x" = "JSON does not conform to schema",
        "i" = "Schema: {.file {schema_path}}",
        "i" = "File: {.file {json_path}}"
      ),
      class = c(error_class, "xpipe_json_schema", "xpipe_json_error", "xpipe_error")
    )
  }

  invisible(TRUE)
}

#' Read JSON Safely
#'
#' @param path Path to JSON file.
#' @param error_title Error title.
#' @param error_class Error classes.
#' @return Parsed JSON as list.
#' @noRd
.read_json_safe <- function(path, error_title, error_class) {
  checkmate::assert_string(path)
  checkmate::assert_string(error_title)
  checkmate::assert_string(error_class)

  if (!fs::file_exists(path)) {
    cli::cli_abort(
      c(error_title, "x" = "JSON file not found: {.file {path}}"),
      class = c(error_class, "xpipe_json_not_found", "xpipe_json_error", "xpipe_error")
    )
  }

  rlang::try_fetch(
    jsonlite::fromJSON(path, simplifyVector = TRUE),
    error = function(e) cli::cli_abort(
      c(error_title, "x" = "Invalid JSON syntax"),
      parent = e,
      class = c(error_class, "xpipe_json_invalid", "xpipe_json_error", "xpipe_error")
    )
  )
}

#' Normalize Configuration Paths
#'
#' Resolves and validates all filesystem paths defined in local.json.
#'
#' Ensures that paths are absolute, normalized, and exist on disk.
#' This function operates purely at the infrastructure level.
#'
#' @param local_config Raw local configuration.
#' @param project_root Base path.
#' @return Normalized local configuration.
#' @noRd
.normalize_paths <- function(local_config, project_root = NULL) {
  if (is.null(project_root)) {
    project_root <- fs::path_abs(".")
  }

  if (is.null(local_config$paths)) {
    cli::cli_abort(
      c("Invalid configuration", "x" = "Missing {.field paths}"),
      class = c("xpipe_local_paths", "xpipe_local_error", "xpipe_error")
    )
  }

  local_config$paths <- lapply(local_config$paths, function(p) {
    p <- fs::path_expand(p)
    p <- if (fs::is_absolute_path(p)) fs::path_norm(p)
    else fs::path_norm(fs::path(project_root, p))

    if (!fs::dir_exists(p)) {
      cli::cli_abort(
        c("Invalid configuration path", "x" = "Path {.file {p}} does not exist"),
        class = c("xpipe_local_paths", "xpipe_local_error", "xpipe_error")
      )
    }

    p
  })

  if (!is.null(local_config$logging$path)) {
    p <- fs::path_norm(fs::path(project_root, local_config$logging$path))
    local_config$logging$path <- p
  }

  local_config
}


#' Validate Batch Layer Presence and Metadata
#'
#' Validates the Batch data layer location and the presence of its
#' persistent metadata file (`layer.json`).
#'
#' The Batch layer is a medallion DATA layer. The associated `layer.json`
#' represents layer STATE (integrity and provenance), not configuration
#' or domain semantics.
#'
#' @param local_config Local configuration.
#' @return Path to layer.json.
#' @noRd
.validate_batch_layer <- function(local_config) {
  batch_path <- local_config$paths$batch

  if (!checkmate::test_string(batch_path, min.chars = 1)) {
    cli::cli_abort(
      c("Invalid configuration", "x" = "Batch layer path not set", "i" = "Expected field: paths.batch"),
      class = c("xpipe_batch_paths", "xpipe_batch_error", "xpipe_error")
    )
  }

  if (!fs::dir_exists(batch_path)) {
    cli::cli_abort(
      c(
        "Invalid Batch layer",
        "x" = "Batch layer directory does not exist",
        "i" = "Path: {.file {batch_path}}"
      ),
      class = c("xpipe_batch_paths", "xpipe_batch_error", "xpipe_error")
    )
  }

  layer_json <- fs::path(batch_path, "layer.json")
  if (!fs::file_exists(layer_json)) {
    cli::cli_abort(
      c(
        "Invalid Batch layer metadata",
        "x" = "Required file {.file layer.json} not found",
        "i" = "Expected: layer.json in {.file {batch_path}}"
      ),
      class = c("xpipe_batch_metadata", "xpipe_batch_error", "xpipe_error")
    )
  }

  .validate_json_schema(
    json_path   = layer_json,
    schema_name = "batch_layer.schema.json",
    error_title = "Invalid Batch layer metadata",
    error_class = "xpipe_batch_metadata"
  )

  layer_json
}

#' Validate Batch Layer Integrity Hash
#'
#' Verifies the integrity of the Batch layer metadata using a shared
#' cryptographic secret. This protects against accidental or unauthorized
#' modification of layer state.
#'
#' @param layer_options Parsed layer.json.
#' @param pepper Runtime key.
#' @param test_string Internal test string.
#' @noRd
.validate_batch_hash <- function(
    layer_options,
    pepper,
    test_string = "byut5f778DfOPth567"
) {
  # Bypass hash check if environment variable is set
  if (Sys.getenv("XPIPE_SKIP_HASH_CHECK") == "TRUE") {
    cli::cli_alert_warning("Bypassing Batch layer integrity hash check.")
    return(invisible(TRUE))
  }

  expected_hash <- layer_options$expected_hash

  if (!checkmate::test_string(expected_hash, min.chars = 1)) {
    cli::cli_abort(
      c("Invalid Batch layer metadata", "x" = "Missing {.field expected_hash} in layer.json"),
      class = c("xpipe_batch_metadata", "xpipe_batch_error", "xpipe_error")
    )
  }

  if (hash_phi(test_string, pepper) != expected_hash) {
    cli::cli_abort(
      c(
        "Invalid Batch layer metadata",
        "x" = "Hash mismatch. System will not start."
      ),
      class = c("xpipe_batch_hash", "xpipe_batch_error", "xpipe_error")
    )
  }

  invisible(TRUE)
}



#' Conditionally print debug information based on local.json
#' @param text The string to print (supports cli formatting)
#' @param .envir The environment to evaluate the string in
#' @keywords internal
#' @noRd
.log_debug <- function(text, .envir = parent.frame()) {
  # 1. Check for manual script overrides first
  is_verbose <- getOption("xpipe.verbose")

  # 2. If no manual override, read from the loaded local.json
  if (is.null(is_verbose)) {
    # Safely retrieve the active config from the package environment
    cfg <- tryCatch(get_xpipe_config(), error = function(e) NULL)

    # Check if the JSON level is explicitly set to "debug"
    is_verbose <- !is.null(cfg) && identical(cfg$local$logging$level, "debug")
  }

  # 3. Print only if debug is enabled
  if (isTRUE(is_verbose)) {
    msg <- c("[DEBUG] ", text)
    cli::cli_inform(msg, .envir = .envir)
  }
}

#' Render Configuration Gracefully
#'
#' Provides a structured console summary of the active control-plane
#' configuration.
#'
#' @param x An S3 object of class \code{xpipe_config}.
#' @param ... Additional arguments (ignored).
#' @noRd
.render_config <- function(x, ...) {
  cli::cli_div(
    theme = list(
      rule = list(color = "cyan", "line-type" = "double")
    )
  )
  cli::cli_rule("Configuration", right = "xpipe")

  pad_nbsp <- function(x, w) gsub(" ", "\u00a0", format(x, width = w))

  # 1. RUNTIME CONTEXT
  cli::cli_h2("Runtime Context")


  cli::cli_bullets(c(
    "{pad_nbsp('Loaded at', 10)}: {.field {format(x$runtime$loaded_at)}}",
    "{pad_nbsp('Security', 10)}: HMAC-SHA256 (Masked Pepper: {.envvar {substr(x$runtime$pepper, 1, 4)}}...)"
  ))


  # 2. LOCAL INFRASTRUCTURE
  cli::cli_h2("Local Infrastructure")
  if (!is.null(x$local$paths)) {
    cli::cli_bullets(
      purrr::imap_chr(
        x$local$paths,
        ~ paste0("{pad_nbsp('", .y, "', 10)}: {.file ", .x, "}")
      )
    )
  }

  # 3. STREAMS CONTRACT (Semantic Truth)
  cli::cli_h2("Streams Contract")
  if (!is.null(x$streams_meta)) {
    cli::cli_bullets(c(
      "{pad_nbsp('Contract type', 13)}: {.field {x$streams_meta$type}}",
      "{pad_nbsp('Version', 13)}: {.field {x$streams_meta$version}}",
      "{pad_nbsp('Maintainer', 13)}: {.field {x$streams_meta$maintainer}}",
      "{pad_nbsp('Last updated', 13)}: {.field {x$streams_meta$last_updated}}"
    ))
  }



  if (!is.null(x$streams)) {
    nms <- names(x$streams)
    labels <- purrr::map_chr(x$streams, "label")
    domains <- purrr::map_chr(x$streams, "schema_domain")
    anchors <- purrr::map_chr(x$streams, "anchor")
    parts <- purrr::map_chr(x$streams, "partition_granularity")

    w_nm  <- max(nchar(nms), nchar("STREAM"))
    w_lbl <- max(nchar(labels), nchar("LABEL"))
    w_dom <- max(nchar(domains), nchar("DOMAIN"))
    w_anc <- max(nchar(anchors), nchar("ANCHOR"))

    # 1. Format the header with ANSI styling via inline formatting
    header <- cli::format_inline(
      "{.strong {format('STREAM', width = w_nm)}   {format('LABEL', width = w_lbl)}   {format('DOMAIN', width = w_dom)}   {format('ANCHOR', width = w_anc)}   PARTITION}"
    )

    # 2. Build the unstyled rows
    table_rows <- purrr::pmap_chr(
      list(nms, labels, domains, anchors, parts),
      function(nm, lbl, dom, anc, part) {
        paste(
          format(nm, width = w_nm),
          format(lbl, width = w_lbl),
          format(dom, width = w_dom),
          format(anc, width = w_anc),
          part,
          sep = "   "
        )
      }
    )

    # 3. Combine and print inside a box
    # cat_boxx handles the printing automatically
    cli::cat_boxx(
      c(header, table_rows),
      header = "Streams",
      border_style = "round",
      padding = c(top = 0, right = 2, bottom = 0, left = 2),
      float = "left",
      border_col = "cyan" # Optional: color the box border itself
    )
  }



  # 4. METHODS CONTRACT (Ingestion Mechanics)
  cli::cli_h2("Methods Contract")
  if (!is.null(x$methods)) {
    nms <- names(x$methods)
    domains <- purrr::map_chr(x$methods, "schema_domain")
    types <- purrr::map_chr(x$methods, "type")

    w_nm  <- max(nchar(nms), nchar("METHOD"))
    w_dom <- max(nchar(domains), nchar("DOMAIN"))

    header <- cli::format_inline(
      "{.strong {format('METHOD', width = w_nm)}   {format('DOMAIN', width = w_dom)}   TYPE}"
    )

    table_rows <- purrr::pmap_chr(
      list(nms, domains, types),
      function(nm, dom, type) {
        paste(
          format(nm, width = w_nm),
          format(dom, width = w_dom),
          type, # Dropping the brackets '[' and ']' since the column header makes it clear
          sep = "   "
        )
      }
    )

    cli::cat_boxx(
      c(header, table_rows),
      header = "Methods",
      border_style = "round",
      padding = c(top = 0, right = 2, bottom = 0, left = 2),
      float = "left",
      border_col = "cyan"
    )
  }

  # 5. BATCH LAYER STATUS
  cli::cli_alert_success("Batch layer metadata validated and trusted.")

  cli::cli_rule()
  cli::cli_end()

  invisible(x)
}

# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
# crypto.R
#
# Pseudonymization and privacy rule application.
#
# Approach: PHI columns are pseudonymized using HMAC-SHA256 with a runtime
# pepper loaded from the environment. The pepper is never stored on disk.
# Pseudonymization is deterministic (same input + pepper = same hash), which
# allows downstream joins without exposing raw identifiers.
#
# Public API:  apply_privacy_rules()
# Internal:    hash_phi(), .pseudonymize_cols()
# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~


# Private API ------------------------------------------------------------------

#' Apply Privacy Rules to an Extracted Data Frame
#'
#' Applies method-level privacy rules to a data frame produced during Batch
#' extraction, before Parquet writing.
#'
#' - `privacy$skip`: drops declared columns (supports `scalar` and `list`)
#' - `privacy$hash`: pseudonymizes declared columns using HMAC-SHA256
#'      (supports `scalar` and `list`, list-columns are hashed element-wise)
#'    The declared shape (`scalar` vs `list`) is authoritative; mismatches abort.
#'
#' Hashing is deterministic (same input + pepper = same hash) and
#' case/whitespace-insensitive, allowing downstream joins without exposing raw
#' identifiers. The pepper is never stored on disk.
#'
#' @param df Data frame to process.
#' @param table_name Character scalar. Name of the extracted table.
#' @param method_config Named list that may contain a `privacy` block.
#'   If `privacy` is absent, the function is a no-op and returns `df` unchanged.
#' @param pepper Character scalar. HMAC key loaded from the environment via
#'   [load_xpipe_config()]. Must be non-empty.
#'
#' @return The input data frame with privacy rules applied.
#'
#' @seealso [hash_phi()] for the underlying hashing primitive.
#' @noRd
apply_privacy_rules <- function(df, table_name, method_config, pepper) {
  checkmate::assert_data_frame(df)
  checkmate::assert_string(table_name)
  checkmate::assert_list(method_config)
  checkmate::assert_string(pepper, min.chars = 1)

  if (nrow(df) == 0) return(df)

  if (is.null(method_config$privacy)) {
    return(df)
  }

  privacy <- method_config$privacy

  .assert_shape <- function(x, shape, tbl, col, action) {
    if (identical(shape, "scalar") && is.list(x)) {
      cli::cli_abort(
        c(
          "Privacy rule shape mismatch",
          "x" = "Column {.val {tbl}.{col}} is a list-column but declared as scalar ({.val {action}})"
        ),
        class = c("xpipe_privacy", "xpipe_batch_error", "xpipe_error")
      )
    }
    if (identical(shape, "list") && !is.list(x)) {
      cli::cli_abort(
        c(
          "Privacy rule shape mismatch",
          "x" = "Column {.val {tbl}.{col}} is scalar but declared as list ({.val {action}})"
        ),
        class = c("xpipe_privacy", "xpipe_batch_error", "xpipe_error")
      )
    }
    if (!identical(shape, "scalar") && !identical(shape, "list")) {
      cli::cli_abort(
        c(
          "Invalid privacy rule",
          "x" = "Unsupported privacy shape {.val {shape}} for {.val {tbl}.{col}}"
        ),
        class = c("xpipe_privacy", "xpipe_batch_error", "xpipe_error")
      )
    }
    invisible(TRUE)
  }

  # 1) SKIP
  skip_spec <- privacy$skip %||% list()
  skip_tbl  <- skip_spec[[table_name]] %||% NULL
  if (!is.null(skip_tbl) && length(skip_tbl) > 0) {
    cols <- intersect(names(skip_tbl), names(df))
    for (col in cols) .assert_shape(df[[col]], skip_tbl[[col]], table_name, col, "skip")
    if (length(cols) > 0) df[cols] <- NULL
  }

  # 2) HASH
  hash_spec <- privacy$hash %||% list()
  hash_tbl  <- hash_spec[[table_name]] %||% NULL
  if (!is.null(hash_tbl) && length(hash_tbl) > 0) {
    cols <- intersect(names(hash_tbl), names(df))
    for (col in cols) {
      shape <- hash_tbl[[col]]
      x <- df[[col]]
      .assert_shape(x, shape, table_name, col, "hash")
      if (identical(shape, "scalar")) {
        df[[col]] <- hash_phi(as.character(x), pepper)
      } else {
        df[[col]] <- lapply(x, function(v) {
          if (is.null(v)) return(NULL)
          if (length(v) == 0) return(character(0))
          hash_phi(as.character(v), pepper)
        })
      }
    }
  }

  df
}



#' HMAC-SHA256 Hash for PHI
#'
#' Hashes a character vector using HMAC-SHA256 keyed by a runtime pepper.
#' Input is uppercased and stripped of leading/trailing whitespace before
#' hashing, making the operation case- and whitespace-insensitive.
#' `NA` and empty strings are returned as `NA_character_`.
#'
#' @param text Character vector of values to hash.
#' @param pepper Character scalar. HMAC key. Must be non-empty.
#'
#' @return Character vector of hex-encoded HMAC-SHA256 hashes,
#'   same length as `text`.
#'
#' @export
#' @keywords internal
hash_phi <- function(text, pepper) {
  if (!checkmate::test_string(pepper, min.chars = 1)) {
    cli::cli_abort(
      "Hashing key (pepper) is missing or empty.",
      class = c("xpipe_runtime_pepper", "xpipe_runtime_error", "xpipe_error")
    )
  }
  pepper_raw <- charToRaw(pepper)

  purrr::map_chr(text, function(x) {
    if (is.na(x) || x == "") return(NA_character_)
    x_clean <- toupper(trimws(as.character(x)))
    as.character(openssl::sha256(charToRaw(x_clean), key = pepper_raw))
  })
}

# Helpers ---------------------------------------------------------------------

#' Pseudonymize Selected Columns
#'
#' @param df Data frame.
#' @param cols Character vector of column names to pseudonymize.
#' @param pepper Character scalar. HMAC key.
#'
#' @return Data frame with selected columns replaced by HMAC-SHA256 hashes.
#'
#' @noRd
.pseudonymize_cols <- function(df, cols, pepper) {
  rlang::check_required(df)
  checkmate::assert_character(cols, min.len = 1)
  checkmate::assert_string(pepper)

  existing <- intersect(cols, names(df))
  if (length(existing) == 0) return(df)
  dplyr::mutate(df, dplyr::across(dplyr::all_of(existing), ~ hash_phi(.x, pepper)))
}

#' Inspect the Batch layer
#'
#' @param config xpipe_config object.
#' @return xpipe_batch_layer
#' @export
inspect_batch_layer <- function(config = get_xpipe_config()) {
  if (!inherits(config, "xpipe_config")) {
    cli::cli_abort(
      "{.arg config} must be a valid {.cls xpipe_config}",
      class = c("xpipe_batch_config", "xpipe_batch_error", "xpipe_error")
    )
  }

  batch_root <- config$local$paths$batch
  if (!fs::dir_exists(batch_root)) {
    cli::cli_abort(
      "Batch layer directory does not exist: {.file {batch_root}}",
      class = c("xpipe_batch_paths", "xpipe_batch_error", "xpipe_error")
    )
  }

  batch_dirs <- fs::dir_ls(batch_root, type = "directory", recurse = FALSE)
  batch_dirs <- batch_dirs[nzchar(basename(batch_dirs))]
  ids <- basename(batch_dirs)

  meta_paths <- fs::path(batch_dirs, "_batch_metadata.json")
  has_meta <- fs::file_exists(meta_paths)

  con <- DBI::dbConnect(duckdb::duckdb(), dbdir = ":memory:", read_only = TRUE)
  on.exit(DBI::dbDisconnect(con, shutdown = TRUE), add = TRUE)

  batches <- vector("list", length(ids))
  names(batches) <- ids

  for (i in seq_along(ids)) {
    bd <- batch_dirs[[i]]
    id <- ids[[i]]

    info <- list(
      batch_id = id,
      path = bd,
      status = "ok",
      meta = NULL,
      parquet_files = character(0),
      parquet_tables = character(0),
      parquet_stats = data.frame(),
      checks = list()
    )

    if (!has_meta[[i]]) {
      info$status <- "incomplete"
      batches[[i]] <- info
      next
    }

    meta <- rlang::try_fetch(
      jsonlite::fromJSON(meta_paths[[i]], simplifyVector = TRUE),
      error = function(e) NULL
    )

    if (is.null(meta)) {
      info$status <- "invalid_metadata"
      batches[[i]] <- info
      next
    }

    info$meta <- meta

    pq_files <- fs::dir_ls(bd, glob = "*.parquet", type = "file")
    info$parquet_files <- as.character(pq_files)
    info$parquet_tables <- fs::path_ext_remove(fs::path_file(pq_files))

    # Stats via DuckDB (no R materialization)
    if (length(pq_files) > 0) {
      stats <- lapply(seq_along(pq_files), function(j) {
        p <- as.character(pq_files[[j]])
        tbl <- info$parquet_tables[[j]]
        p_lit <- DBI::dbQuoteLiteral(con, p)

        ncols <- DBI::dbGetQuery(con, paste0("SELECT COUNT(*) AS n FROM parquet_schema(", p_lit, ")"))$n[[1]]
        nrows <- DBI::dbGetQuery(con, paste0("SELECT COUNT(*) AS n FROM read_parquet(", p_lit, ")"))$n[[1]]

        data.frame(
          table = tbl,
          ncols = as.integer(ncols),
          nrows = as.integer(nrows),
          file = p,
          stringsAsFactors = FALSE
        )
      })
      info$parquet_stats <- do.call(rbind, stats)
    } else {
      info$parquet_stats <- data.frame(
        table = character(),
        ncols = integer(),
        nrows = integer(),
        file = character(),
        stringsAsFactors = FALSE
      )
    }

    # Checks
    method_id <- meta$method$id %||% NA_character_
    method_cfg <- config$methods[[method_id]] %||% NULL
    anchor <- if (!is.null(method_cfg)) method_cfg$anchor_table %||% NA_character_ else NA_character_

    declared_tables <- meta$tables %||% character(0)
    materialized_tables <- info$parquet_tables

    info$checks$anchor_table <- anchor
    info$checks$anchor_materialized <- isTRUE(nzchar(anchor) && anchor %in% materialized_tables)

    info$checks$zero_column_parquet <- if (nrow(info$parquet_stats) > 0) {
      info$parquet_stats$table[info$parquet_stats$ncols == 0]
    } else character(0)

    info$checks$declared_not_materialized <- setdiff(declared_tables, materialized_tables)
    info$checks$materialized_not_declared <- setdiff(materialized_tables, declared_tables)

    batches[[i]] <- info
  }

  structure(
    list(
      root = batch_root,
      count = length(ids),
      batches = batches
    ),
    class = "xpipe_batch_layer"
  )
}



#' Inspect the Bronze layer (DuckDB)
#'
#' Opens the stream DuckDB file in Bronze and reports:
#' - physical tables and row counts
#' - authoritative __promotion_log history
#' - consistency checks between log and physical tables
#'
#' @param stream_id Character scalar. Stream identifier (e.g. "sdo").
#' @param config xpipe_config.
#' @param read_only Logical. Defaults to TRUE.
#'
#' @return An object of class xpipe_bronze_inspection.
#' @export
inspect_bronze <- function(stream_id = NULL, config = get_xpipe_config(), read_only = TRUE) {

  checkmate::assert_string(stream_id, null.ok = TRUE)
  checkmate::assert_class(config, "xpipe_config")
  checkmate::assert_flag(read_only)

  bronze_root <- config$local$paths$bronze
  if (!checkmate::test_string(bronze_root, min.chars = 1)) {
    cli::cli_abort(
      c("Bronze path not configured",
        "x" = "Expected config$local$paths$bronze to be set"),
      class = c("xpipe_bronze_state", "xpipe_error")
    )
  }

  if (is.null(stream_id)) {
    dbs <- fs::dir_ls(bronze_root, glob = "*.duckdb", type = "file")
    streams <- fs::path_ext_remove(fs::path_file(dbs))

    if (length(streams) == 0) {
      cli::cli_abort(
        c("No Bronze databases found",
          "x" = "No *.duckdb files found in {.file {bronze_root}}"),
        class = c("xpipe_bronze_state", "xpipe_error")
      )
    }

    choice <- NULL

    if (interactive() && requireNamespace("climenu", quietly = TRUE)) {
      choice <- tryCatch(
        climenu::select(streams, prompt = "Select Bronze stream to inspect"),
        error = function(e) NULL
      )
    }

    if (!is.null(choice)) {
      stream_id <- choice
    } else {
      # fallback: simple numbered prompt (no cli_inform)
      cat("\nSelect Bronze stream to inspect\n\n")

      for (i in seq_along(streams)) {
        cat(sprintf(" %d. %s\n", i, streams[[i]]))
      }

      cat("\n")
      k <- suppressWarnings(as.integer(readline("Enter number: ")))

      if (is.na(k) || k < 1 || k > length(streams)) {
        cli::cli_abort(
          "Invalid selection.",
          class = c("xpipe_bronze_state", "xpipe_error")
        )
      }

      stream_id <- streams[[k]]
    }

  }

  bronze_db <- fs::path(bronze_root, paste0(stream_id, ".duckdb"))
  if (!fs::file_exists(bronze_db)) {
    cli::cli_abort(
      c("Bronze database not found",
        "x" = "Expected {.file {bronze_db}} to exist"),
      class = c("xpipe_bronze_state", "xpipe_error")
    )
  }

  con <- DBI::dbConnect(
    duckdb::duckdb(),
    dbdir = as.character(bronze_db),
    read_only = read_only
  )
  on.exit(DBI::dbDisconnect(con, shutdown = TRUE), add = TRUE)

  tables <- DBI::dbListTables(con)
  has_log <- "__promotion_log" %in% tables

  # data tables (exclude internal)
  data_tables <- setdiff(tables, "__promotion_log")

  # row counts (can be expensive if huge; acceptable for Bronze inspection)
  row_counts <- if (length(data_tables) > 0) {
    do.call(rbind, lapply(data_tables, function(tbl) {
      n <- DBI::dbGetQuery(
        con,
        paste0("SELECT COUNT(*) AS n FROM ", DBI::dbQuoteIdentifier(con, tbl))
      )$n[[1]]
      data.frame(table_name = tbl, n = as.numeric(n), stringsAsFactors = FALSE)
    }))
  } else {
    data.frame(table_name = character(), n = numeric(), stringsAsFactors = FALSE)
  }

  # promotion history
  promotion_log <- if (isTRUE(has_log)) {
    DBI::dbGetQuery(
      con,
      "SELECT batch_id, stream_id, method_id, promoted_at, batch_created_at, tables, tool, tool_version
       FROM __promotion_log
       ORDER BY promoted_at DESC"
    )
  } else {
    data.frame(
      batch_id = character(),
      stream_id = character(),
      method_id = character(),
      promoted_at = as.POSIXct(character()),
      batch_created_at = as.POSIXct(character()),
      tables = character(),
      tool = character(),
      tool_version = character(),
      stringsAsFactors = FALSE
    )
  }

  # parse promoted table list from JSON
  promoted_tables_from_log <- character(0)
  if (nrow(promotion_log) > 0) {
    promoted_tables_from_log <- unlist(lapply(promotion_log$tables, function(x) {
      if (is.null(x) || is.na(x) || !nzchar(x)) return(character(0))
      out <- tryCatch(jsonlite::fromJSON(x), error = function(e) NULL)
      if (is.null(out)) return(character(0))
      as.character(out)
    }))
    promoted_tables_from_log <- unique(promoted_tables_from_log)
  }

  # sanity checks
  checks <- list()
  checks$has_promotion_log <- isTRUE(has_log)
  checks$log_rows <- nrow(promotion_log)
  checks$data_tables_count <- length(data_tables)

  checks$log_tables_missing_physically <- setdiff(promoted_tables_from_log, data_tables)
  checks$physical_tables_not_in_log <- setdiff(data_tables, promoted_tables_from_log)

  # latest promotion summary
  latest <- if (nrow(promotion_log) > 0) promotion_log[1, , drop = FALSE] else promotion_log

  structure(
    list(
      stream_id = stream_id,
      bronze_db = as.character(bronze_db),
      tables = tables,
      data_tables = data_tables,
      row_counts = row_counts,
      promotion_log = promotion_log,
      latest = latest,
      promoted_tables_from_log = promoted_tables_from_log,
      checks = checks
    ),
    class = "xpipe_bronze_inspection"
  )
}



#' Inspect Silver-current tables
#'
#' Inspects the Silver DuckDB database for one stream/method pair, reporting
#' current tables, row counts, anchor uniqueness, orphan secondary rows, and
#' batch distribution in the current anchor table.
#'
#' @param stream_id Character scalar. Stream identifier.
#' @param method_id Character scalar. Method identifier.
#' @param config xpipe_config.
#' @param read_only Logical. Open Silver database read-only. Defaults to TRUE.
#'
#' @return An object of class `xpipe_silver_current_inspection`.
#' @export
inspect_silver_current <- function(
    stream_id = NULL,
    method_id = NULL,
    config = get_xpipe_config(),
    read_only = TRUE
) {

  checkmate::assert_string(stream_id, null.ok = TRUE)
  checkmate::assert_class(config, "xpipe_config")
  checkmate::assert_flag(read_only)

  if (is.null(stream_id)) {
    streams <- names(config$silver)

    if (length(streams) == 0) {
      cli::cli_abort(
        "No Silver streams configured",
        class = c("xpipe_silver_state", "xpipe_error")
      )
    }

    choice <- NULL
    if (interactive() && requireNamespace("climenu", quietly = TRUE)) {
      choice <- tryCatch(
        climenu::select(streams, prompt = "Select Silver stream to inspect"),
        error = function(e) NULL
      )
    }

    if (!is.null(choice)) {
      stream_id <- choice
    } else {
      cat("\nSelect Silver stream to inspect\n\n")
      for (i in seq_along(streams)) {
        cat(sprintf(" %d. %s\n", i, streams[[i]]))
      }
      cat("\n")
      k <- suppressWarnings(as.integer(readline("Enter number: ")))
      if (is.na(k) || k < 1 || k > length(streams)) {
        cli::cli_abort(
          "Invalid selection.",
          class = c("xpipe_silver_state", "xpipe_error")
        )
      }
      stream_id <- streams[[k]]
    }
  }

  silver_cfg <- config$silver[[stream_id]]
  if (is.null(silver_cfg)) {
    cli::cli_abort(
      c(
        "Silver not configured for stream",
        "x" = "Stream {.val {stream_id}} not found in silver.json"
      ),
      class = c("xpipe_silver_state", "xpipe_error")
    )
  }

  if (!isTRUE(silver_cfg$current$enabled)) {
    cli::cli_abort(
      c(
        "Silver-current disabled",
        "x" = "Silver-current is disabled for stream {.val {stream_id}}"
      ),
      class = c("xpipe_silver_state", "xpipe_error")
    )
  }

  canonical_method <- silver_cfg$canonical_method
  method_id <- method_id %||% canonical_method
  checkmate::assert_string(method_id)

  silver_root <- config$local$paths$silver
  if (!checkmate::test_string(silver_root, min.chars = 1)) {
    cli::cli_abort(
      c(
        "Silver path not configured",
        "x" = "Expected config$local$paths$silver to be set"
      ),
      class = c("xpipe_silver_state", "xpipe_error")
    )
  }

  method_cfg <- config$methods[[method_id]]
  if (is.null(method_cfg)) {
    cli::cli_abort(
      c(
        "Unknown method",
        "x" = "Method {.val {method_id}} not found in methods contract"
      ),
      class = c("xpipe_methods_error", "xpipe_error")
    )
  }

  anchor_tbl <- method_cfg$anchor_table %||% NA_character_
  keys <- method_cfg$identity_keys %||% character(0)

  if (!checkmate::test_string(anchor_tbl, min.chars = 1) ||
      !checkmate::test_character(keys, min.len = 1)) {
    cli::cli_abort(
      c(
        "Invalid Silver configuration",
        "x" = "Method must define anchor_table and identity_keys"
      ),
      class = c("xpipe_silver_config", "xpipe_error")
    )
  }

  silver_db <- fs::path(silver_root, paste0(stream_id, ".duckdb"))
  if (!fs::file_exists(silver_db)) {
    cli::cli_abort(
      c(
        "Silver database not found",
        "x" = "Expected {.file {silver_db}} to exist"
      ),
      class = c("xpipe_silver_state", "xpipe_error")
    )
  }

  con <- DBI::dbConnect(
    duckdb::duckdb(),
    dbdir = as.character(silver_db),
    read_only = read_only
  )
  on.exit(DBI::dbDisconnect(con, shutdown = TRUE), add = TRUE)

  # Silver vs Bronze currency check

  latest_bronze_batch <- NA_character_

  # Try to read Bronze promotion log if Bronze DB is available
  bronze_db <- fs::path(config$local$paths$bronze, paste0(stream_id, ".duckdb"))
  if (fs::file_exists(bronze_db)) {
    try({
      DBI::dbExecute(
        con,
        paste0(
          "ATTACH ",
          DBI::dbQuoteLiteral(con, as.character(bronze_db)),
          " AS bronze (READ_ONLY)"
        )
      )

      latest_bronze_batch <- DBI::dbGetQuery(
        con,
        paste0(
          "SELECT MAX(batch_id) AS batch_id
         FROM bronze.main.__promotion_log
         WHERE stream_id = ",
          DBI::dbQuoteLiteral(con, stream_id),
          " AND method_id = ",
          DBI::dbQuoteLiteral(con, canonical_method)
        )
      )$batch_id[[1]]

      DBI::dbExecute(con, "DETACH bronze")
    }, silent = TRUE)
  }

  latest_silver_batch <- NA_character_
  current_anchor <- paste0("current__", anchor_tbl)

  if (current_anchor %in% DBI::dbListTables(con)) {
    latest_silver_batch <- DBI::dbGetQuery(
      con,
      paste0(
        "SELECT MAX(__batch_id) AS batch_id
       FROM ", DBI::dbQuoteIdentifier(con, current_anchor)
      )
    )$batch_id[[1]]
  }

  silver_status <- dplyr::case_when(
    is.na(latest_bronze_batch) ~ "UNKNOWN",
    is.na(latest_silver_batch) ~ "STALE",
    identical(latest_bronze_batch, latest_silver_batch) ~ "UP TO DATE",
    TRUE ~ "STALE"
  )

  tables <- DBI::dbListTables(con)
  current_tables <- tables[startsWith(tables, "current__")]

  row_counts <- lapply(current_tables, function(tab) {
    sql <- paste0(
      "SELECT ",
      DBI::dbQuoteLiteral(con, tab),
      " AS table_name, COUNT(*) AS n FROM ",
      DBI::dbQuoteIdentifier(con, tab)
    )
    DBI::dbGetQuery(con, sql)
  })

  row_counts <- if (length(row_counts) > 0) {
    do.call(rbind, row_counts)
  } else {
    data.frame(table_name = character(), n = integer())
  }

  current_anchor <- paste0("current__", anchor_tbl)
  anchor_exists <- current_anchor %in% current_tables

  duplicate_keys <- data.frame()
  batch_distribution <- data.frame()
  orphan_checks <- list()

  if (isTRUE(anchor_exists)) {
    anchor_fields <- DBI::dbListFields(con, current_anchor)
    missing_anchor_keys <- setdiff(keys, anchor_fields)

    if (length(missing_anchor_keys) > 0) {
      cli::cli_abort(
        c(
          "Invalid Silver anchor table",
          "x" = "Missing identity key(s): {.val {missing_anchor_keys}}"
        ),
        class = c("xpipe_silver_anchor", "xpipe_error")
      )
    }

    keys_sql <- paste(DBI::dbQuoteIdentifier(con, keys), collapse = ", ")

    duplicate_keys <- DBI::dbGetQuery(
      con,
      paste0(
        "SELECT ", keys_sql, ", COUNT(*) AS n
         FROM ", DBI::dbQuoteIdentifier(con, current_anchor), "
         GROUP BY ", keys_sql, "
         HAVING COUNT(*) > 1
         LIMIT 50"
      )
    )

    batch_distribution <- DBI::dbGetQuery(
      con,
      paste0(
        "SELECT
           __batch_id,
           __batch_created_at,
           COUNT(*) AS n
         FROM ", DBI::dbQuoteIdentifier(con, current_anchor), "
         GROUP BY __batch_id, __batch_created_at
         ORDER BY __batch_created_at DESC, __batch_id DESC"
      )
    )

    secondary_tables <- setdiff(current_tables, current_anchor)

    for (tab in secondary_tables) {
      sec_fields <- DBI::dbListFields(con, tab)
      needed <- c(keys, "__batch_id")

      if (!all(needed %in% sec_fields)) {
        orphan_checks[[tab]] <- data.frame(
          table_name = tab,
          checked = FALSE,
          orphan_rows = NA_integer_,
          reason = "missing identity key(s) or __batch_id"
        )
        next
      }

      join_keys <- paste(
        paste0(
          "s.", DBI::dbQuoteIdentifier(con, keys),
          " = a.", DBI::dbQuoteIdentifier(con, keys)
        ),
        collapse = " AND "
      )

      sql <- paste0(
        "SELECT
           ", DBI::dbQuoteLiteral(con, tab), " AS table_name,
           TRUE AS checked,
           COUNT(*) AS orphan_rows,
           CAST(NULL AS VARCHAR) AS reason
         FROM ", DBI::dbQuoteIdentifier(con, tab), " s
         LEFT JOIN ", DBI::dbQuoteIdentifier(con, current_anchor), " a
           ON ", join_keys, "
          AND s.__batch_id = a.__batch_id
         WHERE a.__batch_id IS NULL"
      )

      orphan_checks[[tab]] <- DBI::dbGetQuery(con, sql)
    }
  }

  orphan_checks <- if (length(orphan_checks) > 0) {
    do.call(rbind, orphan_checks)
  } else {
    data.frame(
      table_name = character(),
      checked = logical(),
      orphan_rows = integer(),
      reason = character()
    )
  }

  structure(
    list(
      stream_id = stream_id,
      method_id = method_id,
      canonical_method = canonical_method,
      silver_db = silver_db,
      tables = tables,
      current_tables = current_tables,
      anchor_table = current_anchor,
      anchor_exists = anchor_exists,
      row_counts = row_counts,
      duplicate_keys = duplicate_keys,
      orphan_checks = orphan_checks,
      batch_distribution = batch_distribution,
      latest_bronze_batch = latest_bronze_batch,
      latest_silver_batch = latest_silver_batch,
      silver_status = silver_status
    ),
    class = "xpipe_silver_current_inspection"
  )
}

#' Print Method for xpipe_batch_layer
#'
#' @export
print.xpipe_batch_layer <- function(x, ...) {
  pad <- function(x, width) {
    x <- as.character(x)
    paste0(x, strrep(" ", pmax(0, width - nchar(x))))
  }

  fmt_bool <- function(v) {
    if (isTRUE(v)) "YES" else "NO"
  }

  head_tail <- function(v, n = 12) {
    v <- as.character(v)
    if (length(v) <= n) return(v)
    c(v[seq_len(n)], paste0("… +", length(v) - n, " more"))
  }

  box_list <- function(v, header, border_col = "cyan") {
    if (length(v) == 0) v <- "None"
    cli::cat_boxx(
      head_tail(v),
      header = header,
      border_style = "round",
      padding = c(top = 0, right = 2, bottom = 0, left = 2),
      float = "left",
      border_col = border_col
    )
  }

  box_table <- function(df, header, border_col = "cyan", max_rows = 20) {
    if (is.null(df) || !is.data.frame(df) || nrow(df) == 0) {
      return(cli::cat_boxx(
        "None",
        header = header,
        border_style = "round",
        padding = c(top = 0, right = 2, bottom = 0, left = 2),
        float = "left",
        border_col = border_col
      ))
    }

    df2 <- df
    if (nrow(df2) > max_rows) df2 <- df2[seq_len(max_rows), , drop = FALSE]

    cols <- names(df2)
    widths <- vapply(cols, function(cn) {
      max(nchar(cn), max(nchar(as.character(df2[[cn]])), na.rm = TRUE))
    }, numeric(1))

    header_row <- paste(pad(cols, widths), collapse = "  ")
    sep_row <- paste(vapply(widths, function(w) strrep("-", w), character(1)), collapse = "  ")

    body <- vapply(seq_len(nrow(df2)), function(i) {
      paste(pad(df2[i, cols, drop = FALSE], widths), collapse = "  ")
    }, character(1))

    lines <- c(header_row, sep_row, body)
    if (nrow(df) > max_rows) {
      lines <- c(lines, paste0("… +", nrow(df) - max_rows, " more"))
    }

    cli::cat_boxx(
      lines,
      header = header,
      border_style = "round",
      padding = c(top = 0, right = 2, bottom = 0, left = 2),
      float = "left",
      border_col = border_col
    )
  }

  cli::cli_h2("Batch layer inspection")

  # Summary counts (supports both old/new shapes)
  batches <- x$batches %||% list()
  ids <- names(batches)

  status_of <- function(b) {
    if (!is.null(b$status)) return(as.character(b$status))
    if (!is.null(b$.status)) return(as.character(b$.status))
    if (is.list(b) && !is.null(b$batch_id)) return("ok")
    "unknown"
  }

  statuses <- vapply(batches, status_of, character(1))
  tab <- sort(table(statuses), decreasing = TRUE)

  summary_rows <- c(
    paste0(pad("Path", 18), ": ", x$root),
    paste0(pad("Batches", 18), ": ", x$count)
  )
  if (length(tab) > 0) {
    for (nm in names(tab)) {
      summary_rows <- c(summary_rows, paste0(pad(paste0("Status ", nm), 18), ": ", tab[[nm]]))
    }
  }

  cli::cat_boxx(
    summary_rows,
    header = "Batch Layer Summary",
    border_style = "round",
    padding = c(top = 0, right = 2, bottom = 0, left = 2),
    float = "left",
    border_col = "cyan"
  )

  if (length(ids) == 0) return(invisible(x))

  cli::cli_h3("Latest batch details")

  # Determine latest batch by metadata execution.created_at when available
  created <- vapply(batches, function(b) {
    m <- b$meta %||% b
    ts <- m$execution$created_at %||% NA_character_
    if (!is.character(ts) || !nzchar(ts)) return(as.POSIXct(NA))
    as.POSIXct(ts, tz = "UTC", tryFormats = c("%Y-%m-%dT%H:%M:%S%z", "%Y-%m-%d %H:%M:%S"))
  }, as.POSIXct(NA))

  latest_id <- if (all(is.na(created))) ids[[1]] else names(which.max(created))
  b <- batches[[latest_id]]
  m <- b$meta %||% b

  method_id <- m$method$id %||% NA_character_
  stream_id <- m$stream$id %||% NA_character_
  created_at <- m$execution$created_at %||% NA_character_

  latest_rows <- c(
    paste0(pad("Batch", 18), ": ", latest_id),
    paste0(pad("Status", 18), ": ", status_of(b)),
    paste0(pad("Stream", 18), ": ", stream_id),
    paste0(pad("Method", 18), ": ", method_id),
    paste0(pad("Created at", 18), ": ", created_at)
  )

  # Extra info if new inspect is used
  if (!is.null(b$parquet_tables)) {
    latest_rows <- c(latest_rows, paste0(pad("Parquet tables", 18), ": ", length(b$parquet_tables)))
  }

  cli::cat_boxx(
    latest_rows,
    header = "Info",
    border_style = "round",
    padding = c(top = 0, right = 2, bottom = 0, left = 2),
    float = "left",
    border_col = "cyan"
  )

  # Show parquet stats if available
  if (!is.null(b$parquet_stats) && is.data.frame(b$parquet_stats)) {
    df <- b$parquet_stats
    keep <- intersect(c("table", "ncols", "nrows"), names(df))
    df <- df[, keep, drop = FALSE]
    box_table(df, header = "Parquet stats", border_col = "cyan", max_rows = 25)
  } else {
    # fallback: show declared tables from metadata if available
    declared <- m$tables %||% character(0)
    box_list(declared, header = "Declared tables (metadata)", border_col = "cyan")
  }

  # Checks (if available)
  if (!is.null(b$checks) && is.list(b$checks)) {
    checks <- b$checks
    anchor <- checks$anchor_table %||% NA_character_
    anchor_ok <- checks$anchor_materialized %||% FALSE
    zero_cols <- checks$zero_column_parquet %||% character(0)
    decl_not_mat <- checks$declared_not_materialized %||% character(0)
    mat_not_decl <- checks$materialized_not_declared %||% character(0)

    sanity_rows <- c(
      paste0(pad("Anchor", 18), ": ", anchor),
      paste0(pad("Anchor ok", 18), ": ", fmt_bool(anchor_ok))
    )


    # build sanity check table
    sanity_df <- data.frame(
      check = c(
        "Anchor table",
        "Zero-column parquet",
        "Declared not materialized",
        "Materialized not declared"
      ),
      result = c(
        if (isTRUE(anchor_ok)) "OK" else "FAIL",
        if (length(zero_cols) == 0) "OK" else "FAIL",
        if (length(decl_not_mat) == 0) "OK" else "WARN",
        if (length(mat_not_decl) == 0) "OK" else "WARN"
      ),
      detail = c(
        anchor %||% "",
        if (length(zero_cols) == 0) "none" else paste(zero_cols, collapse = ", "),
        if (length(decl_not_mat) == 0) "none" else paste(decl_not_mat, collapse = ", "),
        if (length(mat_not_decl) == 0) "none" else paste(mat_not_decl, collapse = ", ")
      ),
      stringsAsFactors = FALSE
    )

    # column widths
    max_check  <- max(nchar(sanity_df$check),  nchar("check"))
    max_result <- max(nchar(sanity_df$result), nchar("result"))
    max_detail <- max(nchar(sanity_df$detail), nchar("detail"))

    sanity_rows <- c(
      paste0(
        pad("check",  max_check),  "  ",
        pad("result", max_result), "  ",
        pad("detail", max_detail)
      ),
      paste0(
        strrep("-", max_check),  "  ",
        strrep("-", max_result), "  ",
        strrep("-", max_detail)
      ),
      vapply(seq_len(nrow(sanity_df)), function(i) {
        paste0(
          pad(sanity_df$check[[i]],  max_check),  "  ",
          pad(sanity_df$result[[i]], max_result), "  ",
          pad(sanity_df$detail[[i]], max_detail)
        )
      }, character(1))
    )

    # overall color: red if any FAIL, yellow if any WARN, else green
    border_col <- if (any(sanity_df$result == "FAIL")) {
      "red"
    } else if (any(sanity_df$result == "WARN")) {
      "yellow"
    } else {
      "green"
    }

    cli::cat_boxx(
      sanity_rows,
      header = "Sanity checks",
      border_style = "round",
      padding = c(top = 0, right = 2, bottom = 0, left = 2),
      float = "left",
      border_col = border_col
    )

  }

  cli::cli_h3("More info")

  # build summary rows for all batches (excluding latest)

  other_ids <- setdiff(names(batches), latest_id)

  other_ids <- other_ids[order(
    vapply(other_ids, function(id) {
      b <- batches[[id]]
      m <- b$meta %||% b
      ts <- m$execution$created_at %||% NA_character_
      as.POSIXct(
        ts,
        tz = "UTC",
        tryFormats = c("%Y-%m-%dT%H:%M:%S%z", "%Y-%m-%d %H:%M:%S")
      )
    }, as.POSIXct(NA)),
    decreasing = TRUE
  )]

  if (length(other_ids) == 0) {
    cli::cat_boxx(
      "No older batches.",
      header = "Batch history (latest first)",
      border_style = "round",
      padding = c(top = 0, right = 2, bottom = 0, left = 2),
      float = "left",
      border_col = "cyan"
    )
  } else {

    rows <- lapply(other_ids, function(id) {
      b <- batches[[id]]
      m <- b$meta %||% b

      stream_id <- m$stream$id %||% ""
      method_id <- m$method$id %||% ""
      created_at <- m$execution$created_at %||% ""

      # human-friendly datetime
      created_at <- tryCatch(
        format(
          as.POSIXct(created_at, tz = "UTC",
                     tryFormats = c("%Y-%m-%dT%H:%M:%S%z", "%Y-%m-%d %H:%M:%S")),
          "%Y-%m-%d %H:%M"
        ),
        error = function(e) created_at
      )

      n_tables <- length(b$parquet_tables %||% character(0))

      # derive generic sanity
      chk <- b$checks %||% list()
      sanity <- "OK"

      if (isFALSE(chk$anchor_materialized) ||
          length(chk$zero_column_parquet %||% character(0)) > 0) {
        sanity <- "FAIL"
      } else if (length(chk$declared_not_materialized %||% character(0)) > 0 ||
                 length(chk$materialized_not_declared %||% character(0)) > 0) {
        sanity <- "WARN"
      }

      c(
        batch_id = id,
        stream   = stream_id,
        method   = method_id,
        created  = created_at,
        tables   = as.character(n_tables),
        sanity   = sanity
      )
    })

    df <- as.data.frame(do.call(rbind, rows), stringsAsFactors = FALSE)

    # column widths
    widths <- vapply(names(df), function(nm) {
      max(nchar(nm), max(nchar(df[[nm]]), na.rm = TRUE))
    }, numeric(1))

    table_rows <- c(
      paste(pad(names(df), widths), collapse = "  "),
      paste(vapply(widths, function(w) strrep("-", w), character(1)), collapse = "  "),
      vapply(seq_len(nrow(df)), function(i) {
        paste(pad(df[i, ], widths), collapse = "  ")
      }, character(1))
    )

    border_col <- if ("FAIL" %in% df$sanity) {
      "red"
    } else if ("WARN" %in% df$sanity) {
      "yellow"
    } else {
      "green"
    }

    cli::cat_boxx(
      table_rows,
      header = "Batch history (latest first)",
      border_style = "round",
      padding = c(top = 0, right = 2, bottom = 0, left = 2),
      float = "left",
      border_col = border_col
    )
  }

  invisible(x)
}

#' Print Method for xpipe_bronze_inspection
#'
#' @export
print.xpipe_bronze_inspection <- function(x, ...) {
  pad <- function(x, width) {
    x <- as.character(x)
    paste0(x, strrep(" ", pmax(0, width - nchar(x))))
  }

  fmt_bool <- function(v) if (isTRUE(v)) "YES" else "NO"

  head_tail <- function(v, n = 12) {
    v <- as.character(v)
    if (length(v) <= n) return(v)
    c(v[seq_len(n)], paste0("… +", length(v) - n, " more"))

  cli::cli_h2("Bronze stream inspection")

    paste0(pad("Database", 18), ": ", x$bronze_db),
    paste0(pad("Tables", 18), ": ", length(x$tables)),
    paste0(pad("Data tables", 18), ": ", length(x$data_tables)),
    paste0(pad("Has promotion log", 18), ": ", fmt_bool(x$checks$has_promotion_log)),
    paste0(pad("Log rows", 18), ": ", x$checks$log_rows)
  )

  cli::cat_boxx(
    summary_rows,
    header = "Bronze Summary",
    border_style = "round",
    padding = c(top = 0, right = 2, bottom = 0, left = 2),
    float = "left",
    border_col = "cyan"
  )

  # Sanity checks table
  missing_phys <- x$checks$log_tables_missing_physically %||% character(0)
  extra_phys <- x$checks$physical_tables_not_in_log %||% character(0)

  sanity_df <- data.frame(
    check = c(
      "Promotion log present",
      "Log tables missing physically",
      "Physical tables not in log"
    ),
    result = c(
      if (isTRUE(x$checks$has_promotion_log)) "OK" else "FAIL",
      if (length(missing_phys) == 0) "OK" else "FAIL",
      if (length(extra_phys) == 0) "OK" else "WARN"
    ),
    detail = c(
      if (isTRUE(x$checks$has_promotion_log)) "__promotion_log" else "missing",
      if (length(missing_phys) == 0) "none" else paste(head_tail(missing_phys), collapse = ", "),
      if (length(extra_phys) == 0) "none" else paste(head_tail(extra_phys), collapse = ", ")
    ),
    stringsAsFactors = FALSE
  )

  max_check  <- max(nchar(sanity_df$check),  nchar("check"))
  max_result <- max(nchar(sanity_df$result), nchar("result"))
  max_detail <- max(nchar(sanity_df$detail), nchar("detail"))

  sanity_rows <- c(
    paste0(pad("check",  max_check),  "  ",
           pad("result", max_result), "  ",
           pad("detail", max_detail)),
    paste0(strrep("-", max_check), "  ",
           strrep("-", max_result), "  ",
           strrep("-", max_detail)),
    vapply(seq_len(nrow(sanity_df)), function(i) {
      paste0(pad(sanity_df$check[[i]],  max_check),  "  ",
             pad(sanity_df$result[[i]], max_result), "  ",
             pad(sanity_df$detail[[i]], max_detail))
    }, character(1))
  )

  border_col <- if (any(sanity_df$result == "FAIL")) "red" else if (any(sanity_df$result == "WARN")) "yellow" else "green"

  cli::cat_boxx(
    sanity_rows,
    header = "Bronze sanity checks",
    border_style = "round",
    padding = c(top = 0, right = 2, bottom = 0, left = 2),
    float = "left",
    border_col = border_col
  )

  # Current tables row counts (top N)
  if (nrow(x$row_counts) > 0) {
    df <- x$row_counts[order(-x$row_counts$n), , drop = FALSE]
    df <- head(df, 20)

    max_t <- max(nchar(df$table_name), nchar("table"))
    max_n <- max(nchar(as.character(df$n)), nchar("rows"))

    rows <- c(
      paste0(pad("table", max_t), "  ", pad("rows", max_n)),
      paste0(strrep("-", max_t), "  ", strrep("-", max_n)),
      vapply(seq_len(nrow(df)), function(i) {
        paste0(pad(df$table_name[[i]], max_t), "  ", pad(df$n[[i]], max_n))
      }, character(1))
    )

    cli::cli_h3("More info")
    cli::cat_boxx(
      rows,
      header = "Top tables by rows",
      border_style = "round",
      padding = c(top = 0, right = 2, bottom = 0, left = 2),
      float = "left",
      border_col = "cyan"
    )
  }

  # Promotion history (other batches)
  if (nrow(x$promotion_log) > 1) {
    df <- x$promotion_log
    df$promoted_at <- as.character(df$promoted_at)
    df$batch_created_at <- as.character(df$batch_created_at)

    df2 <- df[, c("batch_id", "method_id", "batch_created_at", "promoted_at"), drop = FALSE]
    if (nrow(df2) > 20) df2 <- df2[seq_len(20), , drop = FALSE]

    widths <- vapply(names(df2), function(nm) {
      max(nchar(nm), max(nchar(df2[[nm]]), na.rm = TRUE))
    }, numeric(1))

    lines <- c(
      paste(pad(names(df2), widths), collapse = "  "),
      paste(vapply(widths, function(w) strrep("-", w), character(1)), collapse = "  "),
      vapply(seq_len(nrow(df2)), function(i) {
        paste(pad(df2[i, ], widths), collapse = "  ")
      }, character(1))
    )

    cli::cat_boxx(
      lines,
      header = "Promotion history (latest first)",
      border_style = "round",
      padding = c(top = 0, right = 2, bottom = 0, left = 2),
      float = "left",
      border_col = "cyan"
    )
  }

  invisible(x)
}

#' @export
#' Print Silver-current inspection

#' @export

    paste0(x, strrep(" ", pmax(0, width - nchar(x))))
  }

  fmt_bool <- function(x) {
    if (isTRUE(x)) "YES" else "NO"
  }

  cli::cli_h2("Silver-current stream / method inspection")

  summary_rows <- c(

#' Connect to Access database using ODBC (Auto-detect Win/Linux)
#'
#' @param path_to_mdb Path to .mdb file.
#' @return DBI connection object.
#' @keywords internal
#' @noRd
.connect_access_db <- function(path_to_mdb) {
  .mdb_path_abs <- normalizePath(path_to_mdb, winslash = "/", mustWork = TRUE)

  is_windows <- .Platform$OS.type == "windows"

  if (is_windows) {
    driver_name <- "Microsoft Access Driver (*.mdb, *.accdb)"
  } else {
    drivers <- odbc::odbcListDrivers()
    has_mdb <- any(grepl("MDBTools", drivers$name, ignore.case = TRUE))
    if (has_mdb) {
      driver_name <- drivers$name[grep("MDBTools", drivers$name, ignore.case = TRUE)][1]
    } else {
      driver_name <- "/usr/lib/x86_64-linux-gnu/odbc/libmdbodbc.so"
    }
  }

  cli::cli_alert_info("Connecting using driver: {.val {driver_name}}")

  con <- if (is_windows) {
    DBI::dbConnect(
      odbc::odbc(),
      .connection_string = paste0(
        "Driver={", driver_name, "};",
        "DBQ=", .mdb_path_abs, ";"
      )
    )
  } else {
    DBI::dbConnect(
      odbc::odbc(),
      .connection_string = paste0("Driver=MDBTools;DBQ=", .mdb_path_abs, ";"),
      timeout = 0
    )
  }
  return(con)
}

#' Extract Access Database
#'
#' Extracts tables configured in the method source_tables list from an Access Database.
#'
#' @param job Batch job object.
#' @param path Path to .mdb file.
#' @return A named list of data frames.
#' @export
extract_access_db <- function(ctx) {
  is_windows <- .Platform$OS.type == "windows"

  target_tables <- ctx$method_config$source_tables

  if (is.null(target_tables) || length(target_tables) == 0) {
    cli::cli_abort(
      c(
        "No tables defined for extraction",
        "x" = "Method {.val {ctx$method_id}} must define {.field source_tables}"
      ),
      class = c("xpipe_batch_config", "xpipe_batch_error", "xpipe_error")
    )
  }

  path <- ctx$filepath

  if (is.null(target_tables) || length(target_tables) == 0) {
    cli::cli_abort(
      c(
        "No tables defined for extraction",
        "x" = "Method {.val {ctx$method_id}} is missing the {.field target_tables} list."
      ),
      class = c("batch_job", "config")
    )
  }

  if (is_windows) {
    con <- .connect_access_db(path)
    on.exit(DBI::dbDisconnect(con))

    cli::cli_alert_info("Reading {length(target_tables)} tables via ODBC (Windows)...")
    dfs <- list()
    for (t in target_tables) {
      if (t %in% DBI::dbListTables(con)) {
        cli::cli_inform("Reading: {.val {t}}")
        dfs[[t]] <- tibble::as_tibble(DBI::dbReadTable(con, t))
        cli::cli_alert_success("Read {.val {t}} : {nrow(dfs[[t]])} rows")
      }
    }
    return(dfs)

  } else {
    cli::cli_alert_info("Reading {length(target_tables)} tables via mdb-export (Linux)...")

    available_tables <- system2("mdb-tables", args = c("-1", shQuote(path)), stdout = TRUE)

    dfs <- list()
    for (t in target_tables) {
      if (t %in% available_tables) {
        cli::cli_inform("Exporting: {.val {t}}")

        csv_data <- utils::read.csv(pipe(paste("mdb-export", shQuote(path), shQuote(t))), stringsAsFactors = FALSE)
        dfs[[t]] <- tibble::as_tibble(csv_data)

        date_cols <- grep("^d_", names(dfs[[t]]), value = TRUE)
        if (length(date_cols) > 0) {
          cli::cli_inform("Casting {length(date_cols)} date columns...")
          dfs[[t]] <- dplyr::mutate(dfs[[t]], dplyr::across(
            dplyr::all_of(date_cols),
            ~ as.Date(., format = "%m/%d/%y %H:%M:%S")
          ))
        }

        cli::cli_alert_success("Extracted {.val {t}} : {nrow(dfs[[t]])} rows")
      } else {
        cli::cli_alert_warning("Table {.val {t}} not found in .mdb file.")
      }
    }
    return(dfs)
  }
}

# Profile: SDO Tracciato A+B (+C+D)
.xml_profile_sdo_tracciato_abcd <- function(doc, path, ctx) {
  root_name <- xml2::xml_name(xml2::xml_root(doc))

  if (root_name == "flsaInformazioniAnagrafiche") {
    .log_debug("Detected Tracciato A schema.")
    return(list(anagrafica = .parse_tracciato_a(doc)))
  }

#' Extract XML Data (Profile Dispatcher)
#' @export
#' @keywords internal
extract_xml <- function(ctx) {
  checkmate::assert_list(ctx)

  profile <- ctx$method_config$xml_profile %||% NA_character_
  if (!checkmate::test_string(profile, min.chars = 1)) {
    cli::cli_abort(
      c(
        "Invalid method configuration",
        "x" = "Missing {.field xml_profile} for XML method {.val {ctx$method_id}}"
      ),
      class = c("xpipe_batch_extraction", "xpipe_batch_error", "xpipe_error")
    )
  }

  parser <- .xml_profile_registry[[profile]] %||% NULL
  if (is.null(parser) || !is.function(parser)) {
    cli::cli_abort(
      c(
        "Unknown XML profile",
        "x" = "No parser registered for profile {.val {profile}}",
        "i" = "Registered profiles: {.val {names(.xml_profile_registry)}}"
      ),
      class = c("xpipe_batch_extraction", "xpipe_batch_error", "xpipe_error")
    )
  }

  paths <- unique(as.character(unlist(ctx$resolved_inputs)))
  if (length(paths) == 0) {
    paths <- ctx$filepath
  }

  # accumulate per-table outputs across multiple XML files
  acc <- list()

  for (path in paths) {
    cli::cli_inform("Reading XML document: {.file {basename(path)}}")
    doc <- xml2::read_xml(path)

    res <- parser(doc, path, ctx)

    for (nm in names(res)) {
      if (is.null(acc[[nm]])) {
        acc[[nm]] <- res[[nm]]
      } else {
        acc[[nm]] <- .rbind_fill_dfs(list(acc[[nm]], res[[nm]]))
      }
    }
  }

  acc
}



# Helpers ----------------------------------------------------------------

#' Schema-safe row binder (base R)
#'
#' Binds a list of named lists into a data.frame, aligning schemas safely.
#' This is the *canonical* version: predictable, dependency-free, and correct.
#'
#' @param rows List of named lists (each list represents one row)
#' @return data.frame with aligned columns
#' @noRd
.bind_rows_safe <- function(rows) {
  if (length(rows) == 0) return(data.frame())

  all_names <- unique(unlist(lapply(rows, names)))
  rows <- lapply(rows, function(r) {
    missing <- setdiff(all_names, names(r))
    if (length(missing)) r[missing] <- NA_character_
    r[all_names]
  })

  as.data.frame(do.call(rbind, rows))
}

#' Fast row binder with optional data.table acceleration
#'
#' Uses data.table::rbindlist() when available for large batches,
#' otherwise falls back to `.bind_rows_safe()`.
#'
#' @param rows List of named lists
#' @return data.frame
#' @noRd
.bind_rows_fast <- function(rows) {
  if (!length(rows)) return(data.frame())

  if (requireNamespace("data.table", quietly = TRUE)) {
    return(as.data.frame(data.table::rbindlist(rows, fill = TRUE)))
  }

  .bind_rows_safe(rows)
}

#' Build per-record parsing context
#'
#' Extacts and caches DOM information reused across table parsers.
#'
#' @param rec xml2::xml_node representing one <bInformazioniRicovero>
#' @return named list containing cached DOM nodes and attributes
#' @noRd
.get_record_context <- function(rec) {
  rec_children <- xml2::xml_children(rec)
  rec_names    <- xml2::xml_name(rec_children)

  info <- rec_children[rec_names == "informazioniRicovero"][[1]]
  info_children <- xml2::xml_children(info)
  info_names    <- xml2::xml_name(info_children)

  list(
    info          = info,
    info_children = info_children,
    info_names    = info_names,
    codice        = xml2::xml_attr(info, "codiceIstitutoDiCura"),
    prog          = xml2::xml_attr(info, "progressivoSDO"),
    trasm         = rec_children[rec_names == "trasmissione"]
  )
}

# Bind data.frames preserving list-columns, aligning missing columns
.rbind_fill_dfs <- function(dfs) {
  dfs <- dfs[!vapply(dfs, is.null, logical(1))]
  if (length(dfs) == 0) return(NULL)
  if (length(dfs) == 1) return(dfs[[1]])

  all_cols <- unique(unlist(lapply(dfs, names)))
  dfs2 <- lapply(dfs, function(d) {
    missing <- setdiff(all_cols, names(d))
    if (length(missing) > 0) {
      for (m in missing) d[[m]] <- NA
    }
    d[all_cols]
  })
  do.call(rbind, dfs2)
}

#' Parse Tracciato A XML document
#'
#' @param doc xml2::xml_document
#' @return data.frame
#' @noRd
.parse_tracciato_a <- function(doc) {

  records <- xml2::xml_find_all(doc, ".//aInformazioniAnagrafiche")
  n <- length(records)

  rows <- vector("list", n)

  for (i in seq_len(n)) {

    rec <- records[[i]]
    rec_children <- xml2::xml_children(rec)
    rec_names    <- xml2::xml_name(rec_children)

    row <- list()

    trasm <- rec_children[rec_names == "trasmissione"]
    if (length(trasm) == 1) {
      t_children <- xml2::xml_children(trasm[[1]])
      is_scalar  <- xml2::xml_length(t_children) == 0
      if (any(is_scalar)) {
        row[paste0("trasmissione_", xml2::xml_name(t_children[is_scalar]))] <-
          as.list(xml2::xml_text(t_children[is_scalar]))
      }
    }

    info <- rec_children[rec_names == "informazioniAnagrafiche"][[1]]

    row$codiceIstitutoDiCura <- xml2::xml_attr(info, "codiceIstitutoDiCura")
    row$progressivoSDO       <- xml2::xml_attr(info, "progressivoSDO")

    info_children <- xml2::xml_children(info)
    is_scalar <- xml2::xml_length(info_children) == 0
    if (any(is_scalar)) {
      row[xml2::xml_name(info_children[is_scalar])] <-
        as.list(xml2::xml_text(info_children[is_scalar]))
    }

    rows[[i]] <- row
  }

  .bind_rows_fast(rows)
}

#' Parse ricovero row (1:1)
#'
#' @param ctx Record context
#' @return named list
#' @noRd
.parse_b_ricovero <- function(ctx) {

  row <- list()

  if (length(ctx$trasm) == 1) {
    t_children <- xml2::xml_children(ctx$trasm[[1]])
    is_scalar  <- xml2::xml_length(t_children) == 0
    if (any(is_scalar)) {
      row[paste0("trasmissione_", xml2::xml_name(t_children[is_scalar]))] <-
        as.list(xml2::xml_text(t_children[is_scalar]))
    }
  }

  row[names(xml2::xml_attrs(ctx$info))] <- as.list(xml2::xml_attrs(ctx$info))

  is_scalar <- xml2::xml_length(ctx$info_children) == 0
  if (any(is_scalar)) {
    row[ctx$info_names[is_scalar]] <-
      as.list(xml2::xml_text(ctx$info_children[is_scalar]))
  }

  dim <- ctx$info_children[ctx$info_names == "dimissione"]
  if (length(dim) == 1) {
    d_children <- xml2::xml_children(dim[[1]])
    is_scalar  <- xml2::xml_length(d_children) == 0
    if (any(is_scalar)) {
      row[paste0("dim_", xml2::xml_name(d_children[is_scalar]))] <-
        as.list(xml2::xml_text(d_children[is_scalar]))
    }
  }

  row
}

#' Parse diagnosi rows (1–N)
#'
#' @param ctx Record context
#' @return list of named lists
#' @noRd
.parse_b_diagnosi <- function(ctx) {

  out <- list()
  codice <- ctx$codice
  prog   <- ctx$prog

  .normalize_diagnosi_cols <- function(nm) {
    nm <- sub("^diagnosiPrincipaleDimissioneAlRicovero$", "presenteAlRicovero", nm)
    nm <- sub("^diagnosiSecondarieDimissioneAlRicovero$", "presenteAlRicovero", nm)
    nm <- sub("^diagnosiPrincipaleDimissione$", "diagnosiDimissione", nm)
    nm <- sub("^diagnosiSecondarieDimissione$", "diagnosiDimissione", nm)
    nm
  }


  dp <- ctx$info_children[ctx$info_names == "diagnosiPrincipale"]
  if (length(dp) == 1) {
    ch <- xml2::xml_children(dp[[1]])
    is_scalar <- xml2::xml_length(ch) == 0
    if (any(is_scalar)) {
      row <- list(
        codiceIstitutoDiCura = codice,
        progressivoSDO       = prog,
        tipoDiagnosi         = "principale",
        numDiagnosi          = 0L
      )
      nms <- xml2::xml_name(ch[is_scalar])
      nms <- vapply(nms, .normalize_diagnosi_cols, character(1))
      row[nms] <- as.list(xml2::xml_text(ch[is_scalar]))
      out[[length(out) + 1]] <- row
    }
  }

  ds <- ctx$info_children[ctx$info_names == "diagnosiSecondarie"]
  if (length(ds) > 0) {
    for (node in ds) {
      ch <- xml2::xml_children(node)
      is_scalar <- xml2::xml_length(ch) == 0
      if (any(is_scalar)) {
        val <- xml2::xml_attr(node, "numDiagnosiSecondaria")
        num <- if (!is.na(val)) as.integer(val) else NA_integer_
        row <- list(
          codiceIstitutoDiCura = codice,
          progressivoSDO       = prog,
          tipoDiagnosi         = "secondaria",
          numDiagnosi          = num
        )
        nms <- xml2::xml_name(ch[is_scalar])
        nms <- vapply(nms, .normalize_diagnosi_cols, character(1))
        row[nms] <- as.list(xml2::xml_text(ch[is_scalar]))
        out[[length(out) + 1]] <- row
      }
    }
  }

  out
}

#' Parse intervention/procedure rows (1–N)
#'
#' @param ctx Record context
#' @return list of named lists
#' @noRd
.parse_b_procedure <- function(ctx) {

  out <- list()
  codice <- ctx$codice
  prog   <- ctx$prog

  .normalize_proc_cols <- function(nm) {
    nm <- sub("^dataInterventoPrincipale", "dataIntervento", nm)
    nm <- sub("^dataInterventoSecondario", "dataIntervento", nm)
    nm <- sub("^oraInterventoPrincipale", "oraIntervento", nm)
    nm <- sub("^oraInizioInterventoSecondario", "oraIntervento", nm)
    nm <- sub("^chirurgoInterventoPrincipale", "chirurgoIntervento", nm)
    nm <- sub("^chirurgoInterventoSecondario", "chirurgoIntervento", nm)
    nm <- sub("^anestesistaInterventoPrincipale", "anestesistaIntervento", nm)
    nm <- sub("^anestesistaInterventoSecondario", "anestesistaIntervento", nm)
    nm <- sub("^ckListSalaOperatoriaInterventoPrincipale", "ckListSalaOperatoriaIntervento", nm)
    nm <- sub("^ckListSalaOperatoriaInterventoSecondario", "ckListSalaOperatoriaIntervento", nm)
    nm <- sub("^interventoPrincipaleEsterno$", "interventoEsterno", nm)
    nm <- sub("^interventiSecondariEsterni$", "interventoEsterno", nm)
    nm <- sub("^interventoPrincipale", "intervento", nm)
    nm <- sub("^interventiSecondari", "intervento", nm)
    nm
  }

  ip <- ctx$info_children[ctx$info_names == "interventoPrincipale"]
  if (length(ip) == 1) {

    node <- ip[[1]]
    ch <- xml2::xml_children(node)
    is_scalar <- xml2::xml_length(ch) == 0

    row <- list(
      codiceIstitutoDiCura = codice,
      progressivoSDO       = prog,
      tipoIntervento       = "principale",
      numIntervento        = 0L
    )

    if (any(is_scalar)) {
      vals <- xml2::xml_text(ch[is_scalar])
      nms  <- vapply(xml2::xml_name(ch[is_scalar]), .normalize_proc_cols, character(1))

      multi_cols <- c("chirurgoIntervento", "anestesistaIntervento")

      for (nm in unique(nms)) {
        vv <- vals[nms == nm]

        if (nm %in% multi_cols) {
          # sempre list-column, anche se length 1
          row[[nm]] <- list(vv)
        } else {
          # scalare: se ci sono più valori qui è un input sporco → abort
          if (length(vv) > 1) {
            cli::cli_abort(
              c(
                "Unexpected repeated scalar field in procedure",
                "x" = sprintf("Field %s appears %d times in one procedure row", nm, length(vv))
              ),
              class = c("xpipe_batch_extraction", "xpipe_batch_error", "xpipe_error")
            )
          }
          row[[nm]] <- if (length(vv) == 0) NA_character_ else vv[[1]]
        }
      }
    }

    out[[length(out) + 1]] <- row
  }

  sec <- ctx$info_children[ctx$info_names == "interventiSecondari"]
  if (length(sec) > 0) {
    for (node in sec) {

      ch <- xml2::xml_children(node)
      is_scalar <- xml2::xml_length(ch) == 0

      val <- xml2::xml_attr(node, "numInterventoSecondario")
      num <- if (!is.na(val)) as.integer(val) else NA_integer_

      row <- list(
        codiceIstitutoDiCura = codice,
        progressivoSDO       = prog,
        tipoIntervento       = "secondario",
        numIntervento        = num
      )

      if (any(is_scalar)) {
        vals <- xml2::xml_text(ch[is_scalar])
        nms  <- vapply(xml2::xml_name(ch[is_scalar]), .normalize_proc_cols, character(1))

        multi_cols <- c("chirurgoIntervento", "anestesistaIntervento")  # UDI resta scalare

        for (nm in unique(nms)) {
          vv <- vals[nms == nm]

          if (nm %in% multi_cols) {
            row[[nm]] <- list(vv)  # list-column
          } else {
            if (length(vv) > 1) {
              cli::cli_abort(
                c(
                  "Unexpected repeated scalar field in procedure",
                  "x" = sprintf("Field %s appears %d times in one procedure row", nm, length(vv))
                ),
                class = c("xpipe_batch_extraction", "xpipe_batch_error", "xpipe_error")
              )
            }
            row[[nm]] <- if (length(vv) == 0) NA_character_ else vv[[1]]
          }
        }
      }

      out[[length(out) + 1]] <- row
    }
  }

  out
}

#' Parse day-hospital access dates (0–N)
#'
#' @param ctx Record context
#' @return list of named lists
#' @noRd
.parse_b_accessi_dh <- function(ctx) {

  out <- list()
  codice <- ctx$codice
  prog   <- ctx$prog

  dh <- ctx$info_children[ctx$info_names == "AccessiInDH"]
  if (length(dh) == 1) {
    dates <- xml2::xml_children(dh[[1]])
    for (i in seq_along(dates)) {
      out[[length(out) + 1]] <- list(
        codiceIstitutoDiCura = codice,
        progressivoSDO       = prog,
        numAccesso           = i,
        dataAccesso          = xml2::xml_text(dates[[i]])
      )
    }
  }

  out
}

#' Parse transfer rows (0–N)
#'
#' @param ctx Record context
#' @return list of named lists
#' @noRd
.parse_b_trasferimenti <- function(ctx) {

  out <- list()
  codice <- ctx$codice
  prog   <- ctx$prog

  trs <- ctx$info_children[ctx$info_names == "Trasferimenti"]
  if (length(trs) == 0) return(out)

  for (node in trs) {
    num <- xml2::xml_attr(node, "numTrasferimento")
    num <- if (!is.na(num)) as.integer(num) else NA_integer_

    ch <- xml2::xml_children(node)
    is_scalar <- xml2::xml_length(ch) == 0
    if (!any(is_scalar)) next

    row <- list(
      codiceIstitutoDiCura = codice,
      progressivoSDO       = prog,
      numTrasferimento     = num
    )

    row[xml2::xml_name(ch[is_scalar])] <-
      as.list(xml2::xml_text(ch[is_scalar]))

    out[[length(out) + 1]] <- row
  }

  out
}

#' Parse Tracciato B XML document
#'
#' Top-level dispatcher for Tracciato B batch extraction.
#'
#' @param doc xml2::xml_document
#' @return list of data frames
#' @export
.parse_tracciato_b <- function(doc) {

  records <- xml2::xml_find_all(doc, ".//bInformazioniRicovero")
  n <- length(records)

  rows_ricovero   <- vector("list", n)
  rows_diagnosi   <- vector("list", n * 3)
  rows_procedure  <- vector("list", n * 2)
  rows_accessi_dh <- vector("list", n)
  rows_trasferimenti <- vector("list", n * 2)

  kr <- kd <- kp <- kdh <- kt <- 0

  for (rec in records) {
    tryCatch({
      ctx <- .get_record_context(rec)

      ricovero_row <- .parse_b_ricovero(ctx)
      diagnosi_rows <- .parse_b_diagnosi(ctx)
      procedure_rows <- .parse_b_procedure(ctx)
      accessi_rows <- .parse_b_accessi_dh(ctx)
      trasf_rows <- .parse_b_trasferimenti(ctx)

      kr <- kr + 1
      rows_ricovero[[kr]] <- ricovero_row

      for (row in diagnosi_rows)  { kd  <- kd  + 1; rows_diagnosi[[kd]]   <- row }
      for (row in procedure_rows) { kp  <- kp  + 1; rows_procedure[[kp]]  <- row }
      for (row in accessi_rows)   { kdh <- kdh + 1; rows_accessi_dh[[kdh]] <- row }
      for (row in trasf_rows)     { kt  <- kt  + 1; rows_trasferimenti[[kt]] <- row}

    }, error = function(e) {
      cli::cli_warn("Skipping malformed record: {conditionMessage(e)}")
    })
  }

  list(
    ricovero   = .bind_rows_fast(rows_ricovero[seq_len(kr)]),
    diagnosi   = .bind_rows_fast(rows_diagnosi[seq_len(kd)]),
    procedure  = .bind_rows_fast(rows_procedure[seq_len(kp)]),
    accessi_dh = .bind_rows_fast(rows_accessi_dh[seq_len(kdh)]),
    trasferimenti = .bind_rows_fast(rows_trasferimenti[seq_len(kt)])
  )
}
